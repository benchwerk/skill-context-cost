<script>
const secs=[...document.querySelectorAll("section")],bars=[...document.querySelectorAll(".bars li")];
let at=0;
function go(n){
  at=n;
  secs.forEach((s,i)=>s.classList.toggle("show",i===n));
  bars.forEach((b,i)=>{b.classList.toggle("on",i<=Math.min(n,2));b.toggleAttribute("aria-current",i===n)});
  const target=secs[n].querySelector("input:checked,input,h1");
  target.focus({preventScroll:true});scrollTo({top:0});
}
function fail(input,msgId,msg){
  const el=document.getElementById(msgId);el.textContent=msg;
  input.toggleAttribute("aria-invalid",!!msg);if(msg)input.setAttribute("aria-invalid","true");
  return !msg;
}
function validAccount(){
  const e=document.getElementById("email"),p=document.getElementById("pw");
  const okE=fail(e,"e-err",/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(e.value.trim())?"":"Enter an email like name@example.com.");
  const okP=fail(p,"p-err",p.value.length>=8?"":"Your password needs at least 8 characters.");
  if(!okE)e.focus();else if(!okP)p.focus();
  return okE&&okP;
}
document.querySelectorAll("form[data-next]").forEach((f,i)=>f.addEventListener("submit",ev=>{
  ev.preventDefault();
  if(i===0&&!validAccount())return;
  if(i===2){
    const d=new FormData(secs[1].querySelector("form")),g=new FormData(f);
    document.getElementById("summary").textContent=`Welcome. You'll learn ${d.get("lang")} for ${g.get("goal")} a day. Your first lesson is ready.`;
  }
  go(i+1);
}));
document.querySelectorAll("[data-back]").forEach(b=>b.addEventListener("click",()=>go(at-1)));
document.getElementById("restart").addEventListener("click",()=>go(1));
["email","pw"].forEach(id=>document.getElementById(id).addEventListener("input",e=>{
  e.target.removeAttribute("aria-invalid");document.getElementById(id==="pw"?"p-err":"e-err").textContent="";
}));
</script>
