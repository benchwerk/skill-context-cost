const rules={n:v=>v.trim()?'':'Please enter your names.',e:v=>/^\S+@\S+\.\S+$/.test(v)?'':'Please enter a valid email address.',m:v=>v.trim().length>=10?'':'Please write at least a sentence or two.'};
form.addEventListener('submit',ev=>{
  ev.preventDefault();
  let first=null;
  for(const id in rules){
    const msg=rules[id](form.elements[id].value);
    document.getElementById(id+'-e').textContent=msg;
    form.elements[id].setAttribute('aria-invalid',!!msg);
    if(msg&&!first)first=form.elements[id];
  }
  if(first)first.focus();else document.getElementById('box').classList.add('sent');
});
