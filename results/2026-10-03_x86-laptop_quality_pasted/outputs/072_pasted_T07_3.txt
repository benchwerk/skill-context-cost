<script>
  const langs = ["Spanish", "Japanese", "Portuguese", "Korean", "Italian", "Turkish"];
  const goals = [["5 min", "Casual"], ["10 min", "Steady"], ["15 min", "Serious"], ["20 min", "Intense"]];
  const state = { stage: 0, lang: "", goal: "10 min" };
  const $ = (s, r = document) => r.querySelector(s);
  const stages = [...document.querySelectorAll("section")];
  const steps = [...document.querySelectorAll("ol li")];

  const opt = (name, value, sub, checked) =>
    `<label class="opt"><input type="radio" name="${name}" value="${value}" ${checked ? "checked" : ""}><span>${value}${sub ? `<small>${sub}</small>` : ""}</span></label>`;
  $("[data-stage='1'] fieldset").insertAdjacentHTML("beforeend", langs.map(l => opt("lang", l)).join(""));
  $("#goals").insertAdjacentHTML("beforeend", goals.map(([m, s]) => opt("goal", m, s, m === state.goal)).join(""));

  function show(n) {
    state.stage = n;
    stages.forEach((s, i) => s.classList.toggle("active", i === n));
    steps.forEach((li, i) => li.dataset.state = i < n ? "done" : i === n ? "current" : "");
    $("h1", stages[n]).setAttribute("tabindex", "-1");
    $("h1", stages[n]).focus({ preventScroll: true });
    window.scrollTo({ top: 0 });
  }

  function validate(n) {
    if (n === 0) {
      const email = $("#email").value.trim(), pw = $("#pw").value;
      const emailOk = /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email);
      $("#email-err").textContent = emailOk ? "" : "Enter a valid email address.";
      $("#pw-err").textContent = pw.length >= 8 ? "" : "Password must be at least 8 characters.";
      return emailOk && pw.length >= 8;
    }
    if (n === 1) {
      const pick = $("input[name=lang]:checked");
      $("#lang-err").textContent = pick ? "" : "Choose a language to continue.";
      if (pick) state.lang = pick.value;
      return !!pick;
    }
    state.goal = $("input[name=goal]:checked").value;
    return true;
  }

  document.addEventListener("click", e => {
    if (e.target.closest("[data-next]") && validate(state.stage)) {
      if (state.stage === 2) {
        $("#summary-title").textContent = `Welcome, ${$("#email").value.split("@")[0]}.`;
        $("#summary").textContent = `${state.lang} for ${state.goal} a day. Your first lesson is ready.`;
        steps.forEach(li => li.dataset.state = "done");
      }
      show(state.stage + 1);
    }
    if (e.target.closest("[data-back]")) show(state.stage - 1);
  });
  document.addEventListener("keydown", e => {
    if (e.key === "Enter" && e.target.matches("input[type=email], input[type=password]")) $("[data-stage='0'] [data-next]").click();
  });
  show(0);
</script>
