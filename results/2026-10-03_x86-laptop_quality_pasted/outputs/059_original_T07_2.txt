<script>
  const form = document.getElementById('flow');
  const stages = [...form.querySelectorAll('section[data-stage]')];
  const dots = [...document.querySelectorAll('#progress li')];
  let at = 0;

  const rules = [
    () => {
      const bad = {};
      if (!form.name.value.trim()) bad.name = 'Enter your first name.';
      if (!/^\S+@\S+\.\S+$/.test(form.email.value)) bad.email = 'Enter an email like name@example.com.';
      if (form.pw.value.length < 8) bad.pw = 'Use at least 8 characters.';
      return bad;
    },
    () => form.lang.value ? {} : { lang: 'Pick a language to continue.' },
    () => form.goal.value ? {} : { goal: 'Pick a daily goal to finish.' }
  ];

  function show(n) {
    at = n;
    stages.forEach((s, i) => s.classList.toggle('on', i === n));
    dots.forEach((d, i) => d.classList.toggle('on', i <= n));
    const first = stages[n].querySelector('input');
    if (first && n === 0) first.focus({ preventScroll: true });
  }

  function check() {
    const bad = rules[at]();
    stages[at].querySelectorAll('[data-f]').forEach(f => {
      const key = f.dataset.f, msg = bad[key] || '';
      f.classList.toggle('bad', !!msg);
      f.querySelector('.err').textContent = msg;
      const input = f.querySelector('input[type=text],input[type=email],input[type=password]');
      if (input) input.setAttribute('aria-invalid', !!msg);
    });
    const firstBad = Object.keys(bad)[0];
    if (firstBad && form[firstBad] && form[firstBad].focus) form[firstBad].focus();
    return !firstBad;
  }

  form.addEventListener('click', e => {
    if (e.target.matches('[data-next]') && check()) show(at + 1);
    if (e.target.matches('[data-back]')) show(at - 1);
  });
  form.addEventListener('change', e => {
    if (e.target.type === 'radio') stages[at].querySelector('.err').textContent = '';
  });
  form.addEventListener('submit', e => {
    e.preventDefault();
    if (!check()) return;
    const btn = document.getElementById('finish');
    btn.disabled = true; btn.textContent = 'Creating your plan...';
    setTimeout(() => {
      stages.forEach(s => s.classList.remove('on'));
      document.getElementById('done').classList.add('on');
      document.getElementById('who').textContent = form.name.value.trim();
      document.getElementById('sum-lang').textContent = form.lang.value;
      document.getElementById('sum-goal').textContent = form.goal.value + ' minutes a day';
      dots.forEach(d => d.classList.add('on'));
    }, 900);
  });
</script>
