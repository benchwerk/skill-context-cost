const io = new IntersectionObserver(es => es.forEach(e => { if (e.isIntersecting) { e.target.classList.add('in'); io.unobserve(e.target); } }), { threshold: .15 });

const form = document.getElementById('enquiry'), status = document.getElementById('status');
const rules = { name: 'Please enter your names.', email: 'Please enter a valid email address.', date: 'Please choose your wedding date.', msg: 'Please tell me a little about your plans.' };
form.addEventListener('submit', ev => {
  ev.preventDefault(); status.textContent = ''; let ok = true;
  for (const id in rules) {
    const f = form.elements[id], bad = !f.value.trim() || (id === 'email' && !f.checkValidity());
    f.setAttribute('aria-invalid', bad); document.getElementById('e-' + id).textContent = bad ? rules[id] : '';
    if (bad && ok) { f.focus(); ok = false; }
  }
  if (ok) { status.textContent = 'Thank you. I will reply within two working days.'; form.reset(); }
});
