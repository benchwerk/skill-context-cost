<script>
const steps=[...document.querySelectorAll('#steps button')],out=document.getElementById('status');
const reduce=matchMedia('(prefers-reduced-motion:reduce)').matches;
let i=0,timer;
function show(n){i=n;steps.forEach((b,k)=>k===n?b.setAttribute('aria-current','step'):b.removeAttribute('aria-current'));out.textContent=steps[n].dataset.s}
steps.forEach((b,k)=>b.addEventListener('click',()=>{clearInterval(timer);show(k)}));
const io=new IntersectionObserver(es=>es.forEach(e=>{if(e.isIntersecting){e.target.classList.add('on');io.unobserve(e.target)}}),{threshold:.2});
document.querySelectorAll('.rv').forEach(el=>io.observe(el));
if(!reduce){
  const t=new IntersectionObserver(es=>{if(es[0].isIntersecting&&!timer){timer=setInterval(()=>{if(i<3)show(i+1);else clearInterval(timer)},2600);t.disconnect()}},{threshold:.6});
  t.observe(document.getElementById('steps'));
}
