const NAMES=["Mon","Tue","Wed","Thu","Fri","Sat"], CAP=6*60; // booked-minute capacity per day, used for open slot estimate
const now=new Date(), off=(now.getDay()+6)%7, mon=new Date(now);
mon.setDate(now.getDate()-off);
const todayIdx=off>5?0:off; let sel=todayIdx;
const $=id=>document.getElementById(id);
const fmt=d=>d.toLocaleDateString("en-GB",{weekday:"long",day:"numeric",month:"long"});
const dateOf=i=>{const d=new Date(mon);d.setDate(mon.getDate()+i);return d};

const label={done:"Completed",waiting:"Check in done",confirmed:"Check in",unconfirmed:"Confirm by phone"};

function render(){
  const a=D[sel], d=dateOf(sel), open=Math.max(0,Math.round((CAP-mins(a))/30));
  $("title").textContent=sel===todayIdx?"Today":NAMES[sel]+"day".replace("day","");
  $("sub").textContent=fmt(d);
  $("stats").innerHTML=[[a.length,"Appointments"],[(mins(a)/60).toFixed(1),"Chair hours booked"],[a.filter(x=>x[4]==="unconfirmed").length,"Awaiting confirmation"],[open,"Open 30 min slots"]]

};
