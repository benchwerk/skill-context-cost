<script>
  const panels = [...document.querySelectorAll('.panel')];
  const marks = [...document.querySelectorAll('#steps li')];
  const $ = id => document.getElementById(id);
  const form = $('form'), next = $('next'), back = $('back');
  let step = 0;

  const checked = name => form.querySelector(`input[name=${name}]:checked`);

  // Returns true when the current stage is valid, otherwise writes inline errors.
  function validate() {
    if (step === 0) {
      const okMail = /^\S+@\S+\.\S+$/.test($('email').value.trim());
      const okPw = $('pw').value.length >= 8;
      $('e-err').textContent = okMail ? '' : 'Enter a valid email address.';
      $('p-err').textContent = okPw ? '' : 'Password needs at least 8 characters.';
      $('email').setAttribute('aria-invalid', !okMail);
      $('pw').setAttribute('aria-invalid', !okPw);
      return okMail && okPw;
    }
    const name = step === 1 ? 'lang' : 'goal', box = $(name === 'lang' ? 'l-err' : 'g-err');
    const ok = !!checked(name);
    box.textContent = ok ? '' : step === 1 ? 'Choose a language to continue.' : 'Choose a goal to continue.';
    return ok;
  }

  function show(n) {
    step = n;
    panels.forEach((p, i) => p.hidden = i !== n);
    marks.forEach((m, i) => m.classList.toggle('on', i <= Math.min(n, 2)));
    back.hidden = n === 0 || n === 3;
    next.textContent = n === 2 ? 'Start learning' : 'Continue';
    next.hidden = n === 3;
    if (n === 3) $('summary').textContent = `${checked('lang').value}, ${checked('goal').value}. Your first lesson is ready.`;
    panels[n].querySelector('h1').focus({ preventScroll: true });
    window.scrollTo({ top: 0 });
  }

  form.addEventListener('submit', e => { e.preventDefault(); if (validate()) show(step + 1); });
  back.addEventListener('click', () => show(step - 1));
  form.addEventListener('change', e => { if (e.target.type === 'radio') validate(); });
</script>
