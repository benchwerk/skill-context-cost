<script>
  const $ = s => document.querySelector(s), all = s => [...document.querySelectorAll(s)];
  const names = ["Account", "Language", "Daily goal"];
  const langs = [["Spanish","Hola"],["Japanese","Konnichiwa"],["Portuguese","Olá"],["Korean","Annyeong"],["Italian","Ciao"],["Turkish","Merhaba"]];
  const goals = [["5","Casual"],["10","Regular"],["15","Serious"],["20","Intense"]];
  const opt = (name, v, note) => `<label class="opt"><input type="radio" name="${name}" value="${v}"><span>${v}${name==="goal"?" min":""}<small>${note}</small></span></label>`;
  $("#langs").innerHTML = langs.map(([l,h]) => opt("lang", l, h)).join("");
  $("#goals").innerHTML = goals.map(([m,n]) => opt("goal", m, n)).join("");

  let stage = 0;
  const flag = (input, errEl, msg) => { errEl.textContent = msg; input?.setAttribute("aria-invalid", !!msg); return !msg; };
  const checks = [
    () => {
      const okE = flag($("#email"), $("#email-e"), /^\S+@\S+\.\S+$/.test($("#email").value) ? "" : "Enter an email like ana@example.com.");
      const okP = flag($("#pw"), $("#pw-e"), $("#pw").value.length >= 8 ? "" : "Use at least 8 characters.");
      return okE && okP;
    },
    () => flag(null, $("#lang-e"), $("input[name=lang]:checked") ? "" : "Pick a language to continue."),
    () => flag(null, $("#goal-e"), $("input[name=goal]:checked") ? "" : "Pick a daily goal to finish."),
  ];

  function show(n) {
    stage = n;
    all("section").forEach(s => s.hidden = +s.dataset.stage !== n);
    all(".bar i").forEach((b, i) => b.classList.toggle("on", i <= Math.min(n, 2)));
    $("#where").textContent = n < 3 ? names[n] : "Complete";
    $(`section[data-stage="${n}"] h1`).focus({ preventScroll: true });
    scrollTo({ top: 0 });
  }
  all("[data-next]").forEach(b => b.onclick = () => checks[stage]() && show(stage + 1));
  all("[data-back]").forEach(b => b.onclick = () => show(stage - 1));
  all("input[type=radio]").forEach(r => r.onchange = () => checks[stage]());

  $("#flow").onsubmit = e => {
    e.preventDefault();
    if (!checks[2]()) return;
    const btn = $("#finish");
    btn.disabled = true; btn.textContent = "Creating your account...";
    setTimeout(() => {
      $("#summary").textContent = `${$("input[name=lang]:checked").value} for ${$("input[name=goal]:checked").value} minutes a day. Your first lesson is ready.`;
      show(3);
    }, 900);
  };
</script>
