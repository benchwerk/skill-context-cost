const mins = t => { const [h, m] = t.split(":"); return +h * 60 + +m; };
const now = new Date(), cur = now.getHours() * 60 + now.getMinutes();

$("date").textContent = now.toLocaleDateString("en-GB", { weekday: "long", day: "numeric", month: "long", year: "numeric" });
$("s-booked").textContent = appts.length;
$("s-hours").textContent = (appts.reduce((a, r) => a + r[1], 0) / 60).toFixed(1) + " h";

function status(r) {
  const s = mins(r[0]);
  if (cur >= s + r[1]) return ["done", "Completed"];
  if (cur >= s) return ["now", "In the chair"];
  return ["", "Upcoming"];
}

function draw(filter = "") {
  const rows = appts.filter(r => r[2].toLowerCase().includes(filter.toLowerCase().trim()));
  $("list").innerHTML = rows.length ? rows.map((r, i) => {
    const [cls, label] = status(r);
    return `<div class="row ${cls}" style="--i:${i}">
      <span class="time">${r[0]}</span>
      <div><div class="who">${r[2]}</div><div class="what">${r[3]}, ${r[4]}</div></div>
      <span class="tag">${label}</span></div>`;
  }).join("") : `<div class="empty">No patient matches "${filter.replace(/</g, "&lt;")}".<br><button id="clear">Clear search</button></div>`;
}

$("week").innerHTML = week.map(([d, n], i) => `
  <div class="day ${i === 5 ? "today" : ""}" style="--i:${i}">
    <b>${n}</b><i style="height:${n / 12 * 100}%"></i><span>${d}</span></div>`).join("");

$("q").addEventListener("input", e => draw(e.target.value));
$("list").addEventListener("click", e => { if (e.target.id === "clear") { $("q").value = ""; draw(); $("q").focus(); } });
