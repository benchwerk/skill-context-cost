document.documentElement.classList.add('js');
const io=new IntersectionObserver(es=>es.forEach(x=>{if(x.isIntersecting){x.target.classList.add('in');io.unobserve(x.target)}}),{threshold:.15});
document.querySelectorAll('.rv').forEach(el=>io.observe(el));
const f=document.getElementById('f');
f.addEventListener('submit',ev=>{
  ev.preventDefault();
  let ok=true;
  f.querySelectorAll('[required]').forEach(i=>{
    const box=i.closest('.field'),msg=box.querySelector('.err');
    const bad=!i.value.trim()||(i.type==='email'&&!/^\S+@\S+\.\S+$/.test(i.value));
    box.classList.toggle('bad',bad);
    i.setAttribute('aria-invalid',bad);
    msg.textContent=bad?(i.type==='email'?'Enter a valid email address.':'This field is required.'):'';
    if(bad)ok=false;
  });
  if(ok){f.style.display='none';document.getElementById('ok').style.display='block'}
  else f.querySelector('.bad input, .bad textarea').focus();
});
