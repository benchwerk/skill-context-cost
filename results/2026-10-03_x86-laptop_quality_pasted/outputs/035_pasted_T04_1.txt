<script>
  // Sample data for demonstration.
  const CAPACITY = 12;
  const labels = ["Check in", "Mark done", "Completed"];
  const visits = [
    { t: "08:30", n: "Mirela Okonkwo", k: "Cleaning and exam", d: "Dr. Haddad", s: 2 },
    { t: "09:15", n: "Tobias Lindqvist", k: "Root canal, session 2", d: "Dr. Haddad", s: 2 },
    { t: "10:30", n: "Priya Raghunathan", k: "Crown fitting", d: "Dr. Osei", s: 1 },
    { t: "11:15", n: "Gabriel Fontaine", k: "Whitening consult", d: "Dr. Osei", s: 0 },
    { t: "13:00", n: "Ayesha Bandara", k: "Filling, lower left", d: "Dr. Haddad", s: 0 },
    { t: "14:15", n: "Henrik Dvorak", k: "Cleaning and exam", d: "Hygienist Lena", s: 0 },
    { t: "15:30", n: "Noor Castellanos", k: "Orthodontic review", d: "Dr. Osei", s: 0 },
  ];
  const week = [["Mon", 10], ["Tue", 11], ["Wed", 8], ["Thu", 12], ["Fri", 9], ["Sat", 5]];
  const $ = id => document.getElementById(id);

  function renderList() {
    $("list").innerHTML = visits.map((v, i) => `
      <div class="row ${v.s === 2 ? "done" : ""}">
        <span class="time">${v.t}</span>
        <div class="who"><strong>${v.n}</strong><span>${v.k}, ${v.d}</span></div>
        <button data-i="${i}" data-s="${v.s}" ${v.s === 2 ? "disabled" : ""}>${labels[v.s]}</button>
      </div>`).join("");
    $("s-total").textContent = visits.length;
    $("s-in").textContent = visits.filter(v => v.s === 1).length;
    $("s-done").textContent = visits.filter(v => v.s === 2).length;
    $("s-open").textContent = CAPACITY - visits.length;
  }

  $("list").addEventListener("click", e => {
    const b = e.target.closest("button");
    if (!b || visits[b.dataset.i].s >= 2) return;
    visits[b.dataset.i].s++;
    renderList();
  });

  const now = new Date();
  $("today").textContent = now.toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric", year: "numeric" });
  const todayName = now.toLocaleDateString("en-US", { weekday: "short" });
  $("week").innerHTML = week.map(([d, n]) => `
    <div class="day ${d === todayName ? "today" : ""}">
      <em>${n}</em><div class="bar" style="height:${(n / CAPACITY) * 100}%"></div>${d}
    </div>`).join("");
  const booked = week.reduce((a, w) => a + w[1], 0);
  $("wknote").textContent = `${booked} of ${CAPACITY * week.length} slots booked this week. Thursday is full.`;
  renderList();
</script>
