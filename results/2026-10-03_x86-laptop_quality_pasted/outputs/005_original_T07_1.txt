<script>
const $=s=>document.querySelector(s), panels=[...document.querySelectorAll('section')], dots=[...$('#bar').children];
let cur=0;
function show(n){
  cur=n;
  panels.forEach((p,i)=>p.hidden=i!==n);
  dots.forEach((d,i)=>d.classList.toggle('on',i<=Math.min(n,2)));
  $('#back').hidden=n===0||n===3;
  if(n===3)$('#bar').hidden=true;
  panels[n].querySelector('h1').setAttribute('tabindex','-1');
  panels[n].querySelector('h1').focus({preventScroll:false});
}
function fail(id,msg){const i=$('#'+id),e=$('#e-'+id);if(i)i.setAttribute('aria-invalid',!!msg);e.textContent=msg||'';return !msg}
function valid(n){
  if(n===0){
    const a=fail('name',$('#name').value.trim()?'':'Enter your first name.');
    const b=fail('email',/^\S+@\S+\.\S+$/.test($('#email').value)?'':'Enter a valid email, like sam@example.com.');
    const c=fail('pw',$('#pw').value.length>=8?'':'Use at least 8 characters.');
    if(!(a&&b&&c))panels[0].querySelector('[aria-invalid=true]').focus();
    return a&&b&&c;
  }
  if(n===1)return fail('lang',document.querySelector('[name=lang]:checked')?'':'Pick a language to continue.');
  return fail('goal',document.querySelector('[name=goal]:checked')?'':'Pick a goal to continue.');
}
$('#f').addEventListener('submit',e=>{
  e.preventDefault();
  if(!valid(cur))return;
  if(cur<2)return show(cur+1);
  const b=$('#finish');b.disabled=true;b.textContent='Creating your plan...';
  setTimeout(()=>{
    $('#doneH').textContent='Welcome, '+$('#name').value.trim()+'.';
    $('#doneP').textContent=document.querySelector('[name=goal]:checked').value+' minutes of '+document.querySelector('[name=lang]:checked').value+' a day. Your first lesson is ready.';
    show(3);
  },900);
});
$('#back').addEventListener('click',()=>show(cur-1));
document.querySelectorAll('[name=lang],[name=goal]').forEach(r=>r.addEventListener('change',()=>{$('#e-lang').textContent=$('#e-goal').textContent=''}));
</script>
