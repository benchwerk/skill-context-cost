<script>
  // Reveal on scroll: shows sections in reading order as they arrive (hierarchy).
  const io = new IntersectionObserver((entries) => {
    entries.forEach((e) => { if (e.isIntersecting) { e.target.classList.add("in"); io.unobserve(e.target); } });
  }, { threshold: 0.2 });
  document.querySelectorAll(".reveal").forEach((el) => io.observe(el));

  // Signup feedback: inline error below the field, confirmation on success.
  const form = document.getElementById("signup"), input = document.getElementById("email"), msg = document.getElementById("msg");
  form.addEventListener("submit", (e) => {
    e.preventDefault();
    const ok = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(input.value.trim());
    input.setAttribute("aria-invalid", String(!ok));
    msg.textContent = ok ? "You are on the list. See you on launch day." : "Enter a valid email address, like name@example.com.";
  });
</script>
