<script>
  // Sample data for illustration
  const appts = [
    { t: "08:30", n: "Amara Okonkwo-Reyes", k: "Hygiene clean", m: 45, in: true },
    { t: "09:15", n: "Tobias Lindqvist", k: "Crown fitting", m: 60, in: true },
    { t: "10:30", n: "Priya Venkataraghavan", k: "Root canal, visit 2", m: 90, in: false },
    { t: "11:00", n: "Declan Murchison", k: "Check-up", m: 30, in: false },
    { t: "13:30", n: "Noor Al-Hashimi", k: "Whitening consult", m: 30, in: false },
    { t: "14:15", n: "Wiremu Tane", k: "Filling, lower left", m: 45, in: false },
    { t: "15:45", n: "Saoirse Quigley", k: "Orthodontic review", m: 30, in: false },
  ];
  const weekCounts = [9, 11, 7, appts.length, 12, 4];
  const names = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
  let filter = "all";
  const $ = id => document.getElementById(id);
  const initials = n => n.split(" ").map(w => w[0]).slice(0, 2).join("");

  function renderStats() {
    const done = appts.filter(a => a.in).length;
    const chair = appts.reduce((s, a) => s + a.m, 0);
    const rows = [
      [appts.length, "Appointments today"],
      [done, "Checked in"],
      [(chair / 60).toFixed(1) + " h", "Chair time booked"],
      [weekCounts.reduce((a, b) => a + b, 0), "Booked this week"],
    ];
    $("stats").innerHTML = rows.map(([v, l]) => `<div class="stat"><b>${v}</b><span>${l}</span></div>`).join("");
  }
  function renderList() {
    const shown = appts.filter(a => filter === "all" || (filter === "in") === a.in);
    if (!shown.length) {
      $("list").innerHTML = `<div class="empty">No patients here yet. Check someone in from the All tab.</div>`;
      return;
    }
    const row = a => `<button type="button" class="appt ${a.in ? "in" : ""}" data-t="${a.t}">
      <span class="time">${a.t}</span><span class="av" aria-hidden="true">${initials(a.n)}</span>
      <span class="who">${a.n}<small>${a.k}, ${a.m} min</small></span>
      <span class="tag">${a.in ? "Checked in" : "Waiting"}</span></button>`;
    const part = (label, list) => list.length ? `<div class="group"><h3>${label}</h3>${list.map(row).join("")}</div>` : "";
    $("list").innerHTML = part("Morning", shown.filter(a => a.t < "12:00")) + part("Afternoon", shown.filter(a => a.t >= "12:00"));
  }
  function renderWeek() {
    const max = Math.max(...weekCounts);
    const dow = (new Date().getDay() + 6) % 7; // Monday = 0, Sunday = 6 (no highlight)
    $("week").innerHTML = weekCounts.map((c, i) =>
      `<div class="day ${i === dow ? "today" : ""}"><em>${c}</em><div class="bar" style="--i:${i};height:${Math.round(c / max * 120)}px"></div>${names[i]}</div>`).join("");
  }
  $("today").textContent = new Date().toLocaleDateString("en-GB", { weekday: "long", day: "numeric", month: "long" }) + " with Dr. Fiona Calloway";
  $("list").addEventListener("click", e => {
    const b = e.target.closest(".appt");
    if (!b) return;
    const a = appts.find(x => x.t === b.dataset.t);
    a.in = !a.in;
    renderList(); renderStats();
  });
  document.querySelector(".tabs").addEventListener("click", e => {
    const b = e.target.closest("button");
    if (!b) return;
    filter = b.dataset.f;
    document.querySelectorAll(".tabs button").forEach(x => x.setAttribute("aria-pressed", x === b));
    renderList();
  });
  renderStats(); renderList(); renderWeek();
</script>
</body>
