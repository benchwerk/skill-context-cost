<script>
  // Scroll reveal (hierarchy) and the tracker lighting up in sequence (storytelling).
  const reduce = matchMedia("(prefers-reduced-motion: reduce)").matches;
  const io = new IntersectionObserver(entries => entries.forEach(e => {
    if (e.isIntersecting) { e.target.classList.add("in"); io.unobserve(e.target); }
  }), { threshold: 0.2 });
  document.querySelectorAll(".rv").forEach(el => io.observe(el));

  const steps = [...document.querySelectorAll("#steps li")];
  const play = () => steps.forEach((li, i) => setTimeout(() => li.classList.add("done"), reduce ? 0 : i * 700));
  new IntersectionObserver((entries, obs) => {
    if (entries[0].isIntersecting) { play(); obs.disconnect(); }
  }, { threshold: 0.5 }).observe(document.getElementById("steps"));
</script>
