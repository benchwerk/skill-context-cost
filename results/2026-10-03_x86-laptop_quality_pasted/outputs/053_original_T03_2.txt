const io = new IntersectionObserver((es) => es.forEach((x) => {
  if (x.isIntersecting) { x.target.classList.add('in'); io.unobserve(x.target); }
}), { threshold: .15 });
document.querySelectorAll('.reveal').forEach((el) => io.observe(el));

const f = document.getElementById('f');
const rules = {
  n: (v) => v.trim() ? '' : 'Please enter your names.',
  e: (v) => /^\S+@\S+\.\S+$/.test(v) ? '' : 'Enter a valid email, like name@example.com.',
  d: (v) => v ? '' : 'Choose an approximate wedding date.',
  m: (v) => v.trim().length > 9 ? '' : 'Add a sentence or two about your day.'
};
function check(id) {
  const el = document.getElementById(id), msg = rules[id](el.value);
  document.getElementById(id + '-e').textContent = msg;
  el.setAttribute('aria-invalid', msg ? 'true' : 'false');
  return !msg;
}
Object.keys(rules).forEach((id) => document.getElementById(id).addEventListener('blur', () => check(id)));
f.addEventListener('submit', (ev) => {
  ev.preventDefault();
  const bad = Object.keys(rules).filter((id) => !check(id));
  if (bad.length) { document.getElementById(bad[0]).focus(); return; }
  const b = f.querySelector('button');
  b.disabled = true; b.textContent = 'Sending...';
  setTimeout(() => { f.hidden = true; const ok = document.getElementById('ok'); ok.hidden = false; ok.focus(); }, 900);
});
