const appts = [
  {t:"08:30", n:"Marguerite Oyelaran", r:"Check-up and clean", d:"Okafor", m:45},

  {t:"15:00", n:"Ines Brandvold", r:"Extraction review", d:"Lindqvist", m:30}
];
const week = [["Mon",11],["Tue",13],["Wed",10],["Thu",9],["Fri",12],["Sat",7]];
const labels = ["Arrived", "In chair", "Done"];
const state = appts.map((_, i) => (i === 0 ? 2 : i === 1 ? 1 : 0));
const $ = id => document.getElementById(id);

function render() {
  $("list").innerHTML = appts.map((a, i) => `
    <li class="appt ${state[i] === 2 ? "done" : ""}">
      <span class="time">${a.t}</span>
      <div><div class="name">${a.n}</div><div class="mute">${a.r}, Dr. ${a.d}, ${a.m} min</div></div>
      <button data-i="${i}" data-s="${state[i]}" aria-label="${a.n}: ${state[i] ? labels[state[i]] : "Not arrived"}. Advance status">${state[i] ? labels[state[i]] : "Check in"}</button>
    </li>`).join("");
  const arrived = state.filter(s => s > 0).length;
  $("s-total").textContent = appts.length;
  $("s-in").textContent = arrived;
  $("s-left").textContent = appts.length - arrived;
  $("s-hrs").textContent = (appts.reduce((x, a) => x + a.m, 0) / 60).toFixed(1) + " h";
}

$("list").addEventListener("click", e => {
  const b = e.target.closest("button");
  if (!b) return;
  const i = +b.dataset.i;
  state[i] = (state[i] + 1) % 3;
  render();
  $("list").querySelector(`button[data-i="${i}"]`).focus();
});

const today = new Date();
$("date").textContent = today.toLocaleDateString("en-GB", {weekday: "long", day: "numeric", month: "long"});
const max = Math.max(...week.map(w => w[1]));
const todayName = today.toLocaleDateString("en-US", {weekday: "short"});
$("week").innerHTML = week.map(([d, c]) => `
  <div class="day ${d === todayName ? "today" : ""}">
    <em>${c}</em><i style="height:${(c / max) * 100 * 0.78}%"></i><span>${d}</span>
  </div>`).join("");

render();
</script>
