<script>
const ICON = {
  done:  '<svg viewBox="0 0 16 16" fill="none" stroke="var(--ok)" stroke-width="2"><circle cx="8" cy="8" r="6.5"/><path d="M5 8.3l2 2 4-4.3"/></svg>',
  chair: '<svg viewBox="0 0 16 16" fill="var(--info)"><circle cx="8" cy="8" r="6.5"/><path d="M6.3 5l5 3-5 3z" fill="var(--card)"/></svg>',
  wait:  '<svg viewBox="0 0 16 16" fill="none" stroke="var(--warn)" stroke-width="2"><circle cx="8" cy="8" r="6.5"/><path d="M8 4.5V8l2.3 1.5"/></svg>',
  conf:  '<svg viewBox="0 0 16 16" fill="none" stroke="var(--muted)" stroke-width="2"><circle cx="8" cy="8" r="6.5"/></svg>',
  cancel:'<svg viewBox="0 0 16 16" fill="none" stroke="var(--crit)" stroke-width="2"><circle cx="8" cy="8" r="6.5"/><path d="M5.5 5.5l5 5M10.5 5.5l-5 5"/></svg>'
};
const LABEL = { done:'Completed', chair:'In chair', wait:'Waiting', conf:'Confirmed', cancel:'Cancelled' };
const appts = [
  ['08:30','Maria Alvarez','Check-up & cleaning','done'],
  ['09:15','Tom Becker','Filling (upper left)','done'],
  ['10:00','Aisha Khan','Root canal, visit 2','chair'],
  ['10:45','Liam O\'Connor','Whitening consult','wait'],
  ['11:30','Priya Nair','Crown fitting','conf'],
  ['13:00','Hannah Weiss','Child check-up','conf'],
  ['13:45','Diego Santos','Extraction','cancel'],
  ['14:30','Grace Lee','Braces adjustment','conf'],
  ['15:15','Omar Farouk','Scale & polish','conf']
];
const week = [9, 11, 8, 10, 7, 4];            // Mon–Sat booked slots
const CAP = 12, names = ['Mon','Tue','Wed','Thu','Fri','Sat'];
const full = ['Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
const now = new Date(), todayIdx = (now.getDay() + 6) % 7;   // Mon=0

document.getElementById('date').textContent =
  now.toLocaleDateString(undefined, { weekday:'long', day:'numeric', month:'long', year:'numeric' });

const active = appts.filter(a => a[3] !== 'cancel');
const doneN = appts.filter(a => a[3] === 'done').length;
const stats = [
  [active.length, 'Appointments today', appts.length - active.length + ' cancelled'],
  [doneN + ' / ' + active.length, 'Completed', appts.filter(a => a[3]==='chair'||a[3]==='wait').length + ' in clinic now'],
  [CAP - active.length, 'Open slots today', 'Next free: 16:00'],
  [week.reduce((a,b)=>a+b,0), 'Booked this week', Math.round(week.reduce((a,b)=>a+b,0) / (CAP*week.length) * 100) + '% of capacity']
];
document.getElementById('stats').innerHTML = stats.map(s =>
  `<div class="card stat"><div class="l">${s[1]}</div><div class="v">${s[0]}</div><div class="n">${s[2]}</div></div>`).join('');

document.getElementById('count').textContent = appts.length + ' booked';
document.getElementById('sched').innerHTML = appts.map(a =>
  `<tr class="${a[3]==='chair'?'now':''}"><td class="t">${a[0]}</td><td class="pt">${a[1]}</td><td>${a[2]}</td>` +
  `<td><span class="badge">${ICON[a[3]]}${LABEL[a[3]]}</span></td></tr>`).join('');

const tip = document.getElementById('tip');
document.getElementById('chart').innerHTML = week.map((n, i) =>
  `<div class="col ${i===todayIdx?'today':''}" tabindex="0" data-i="${i}" aria-label="${full[i]}: ${n} of ${CAP} slots booked">` +
  `<span class="val">${n}</span><div class="bar" style="height:${n/CAP*100*0.82}%"></div></div>`).join('');
document.getElementById('days').innerHTML = names.map((d, i) => `<span class="${i===todayIdx?'today':''}">${d}</span>`).join('');
document.querySelectorAll('.col').forEach(c => {
  const show = e => {
    const i = +c.dataset.i, r = c.getBoundingClientRect();
    tip.innerHTML = `<b>${full[i]}</b> · ${week[i]} booked, ${CAP - week[i]} open`;
    tip.style.left = Math.max(8, r.left + r.width/2 - tip.offsetWidth/2) + 'px';
    tip.style.top = (r.top - 34) + 'px'; tip.style.opacity = 1;
  };
  c.addEventListener('mouseenter', show); c.addEventListener('focus', show);
  c.addEventListener('mouseleave', () => tip.style.opacity = 0); c.addEventListener('blur', () => tip.style.opacity = 0);
});

document.querySelector('#wk-table tbody').innerHTML = week.map((n, i) =>
  `<tr><td>${full[i]}</td><td>${n}</td><td>${CAP - n}</td></tr>`).join('');
const btn = document.getElementById('toggle');
btn.onclick = () => {
  const t = btn.getAttribute('aria-pressed') === 'false';
  btn.setAttribute('aria-pressed', t); btn.textContent = t ? 'Chart view' : 'Table view';
  document.getElementById('wk-chart').style.display = t ? 'none' : 'block';
  document.getElementById('wk-table').style.display = t ? 'table' : 'none';
};
</script>
