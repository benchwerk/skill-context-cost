<script>
  const form = document.getElementById('rsvp'), status = document.getElementById('status');
  form.date.min = new Date().toISOString().split('T')[0];
  form.addEventListener('submit', e => {
    e.preventDefault();
    const bad = [...form.querySelectorAll('[required]')].find(f => !f.value.trim() || !f.checkValidity());
    if (bad) { status.textContent = 'Please complete the ' + bad.name + ' field.'; bad.focus(); return; }
    const d = new Date(form.date.value + 'T12:00');
    if (d.getDay() === 1) { status.textContent = 'We are closed on Mondays. Please pick another day.'; form.date.focus(); return; }
    status.textContent = `Thank you, ${form.name.value.split(' ')[0]}. We will confirm ${form.guests.value} at ${form.time.value} by email.`;
    form.reset(); form.date.min = new Date().toISOString().split('T')[0];
  });
  const io = new IntersectionObserver(es => es.forEach(x => x.isIntersecting && (x.target.classList.add('in'), io.unobserve(x.target))), { threshold: .15 });
  document.querySelectorAll('.reveal').forEach(el => io.observe(el));
</script>
