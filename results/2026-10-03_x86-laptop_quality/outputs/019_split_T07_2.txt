<script>
  const langs = [["Spanish", "Español"], ["French", "Français"], ["Japanese", "日本語"], ["German", "Deutsch"], ["Italian", "Italiano"], ["Korean", "한국어"]];
  const goals = [[5, "Casual", "One short lesson"], [10, "Regular", "A lesson and a review"], [15, "Serious", "Two lessons, steady progress"], [30, "Intense", "Fluency sprint"]];
  const $ = (s, r = document) => r.querySelector(s);
  const steps = [...document.querySelectorAll("fieldset")], bars = [...document.querySelectorAll(".progress i")];
  const form = $("form"), cta = $(".cta"), back = $(".back"), err = $(".err");
  let step = 0;

  const card = (name, value, title, sub, cls = "") => `<div class="opt"><input type="radio" name="${name}" id="${name}-${value}" value="${value}">
    <label for="${name}-${value}" class="${cls}"><b>${title}</b><small>${sub}</small></label></div>`;
  $(".two").innerHTML = langs.map(([l, n]) => card("lang", l, l, n)).join("");
  $(".goals").innerHTML = goals.map(([m, t, d]) => card("goal", m, `${t}`, d, "goal").replace(`<b>${t}</b>`, `<b>${t}</b><em>${m} min</em>`)).join("");

  function show(n) {
    step = n;
    steps.forEach((s, i) => s.classList.toggle("active", i === n));
    bars.forEach((b, i) => b.classList.toggle("on", i <= n));
    back.hidden = n === 0;
    cta.textContent = ["Continue", "Continue", "Start learning"][n];
    steps[n].querySelector("input")?.focus({ preventScroll: true });
  }

  function valid() {
    const f = new FormData(form);
    if (step === 0) {
      const okMail = /^\S+@\S+\.\S+$/.test(f.get("email") || ""), okPw = (f.get("pw") || "").length >= 8;
      form.email.setAttribute("aria-invalid", !okMail); form.pw.setAttribute("aria-invalid", !okPw);
      err.textContent = !okMail ? "Enter a valid email address." : !okPw ? "Use at least 8 characters for your password." : "";
      return okMail && okPw;
    }
    return !!f.get(step === 1 ? "lang" : "goal");
  }

  form.addEventListener("submit", e => {
    e.preventDefault();
    if (!valid()) { if (step) cta.animate([{ transform: "translateX(-6px)" }, { transform: "translateX(6px)" }, { transform: "none" }], 220); return; }
    if (step < 2) return show(step + 1);
    const f = new FormData(form);
    form.hidden = true; $(".progress").hidden = true; back.hidden = true;
    const done = $(".done"); done.hidden = false;
    $("h1", done).textContent = `${f.get("goal")} minutes of ${f.get("lang")} a day`;
    $(".lede", done).textContent = "Your first lesson is ready. We'll nudge you at the same time each day.";
  });
  back.addEventListener("click", () => show(step - 1));
  show(0);
</script>
