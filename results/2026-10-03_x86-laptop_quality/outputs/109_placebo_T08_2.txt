  const fmt = (el, v) => {
    const d = +el.dataset.dec || 0, p = el.dataset.prefix || "", s = el.dataset.suffix || "";
    el.textContent = p + v.toLocaleString("en-US", { minimumFractionDigits: d, maximumFractionDigits: d }) + s;
  };
  const io = new IntersectionObserver(entries => entries.forEach(e => {
    if (!e.isIntersecting) return;
    io.unobserve(e.target);
    const el = e.target, to = +el.dataset.to, t0 = performance.now();
    const reduce = matchMedia("(prefers-reduced-motion: reduce)").matches;
    (function tick(t) {
      const k = reduce ? 1 : Math.min((t - t0) / 1400, 1);
      fmt(el, to * (1 - Math.pow(1 - k, 3)));
      if (k < 1) requestAnimationFrame(tick);
    })(t0);
  }), { threshold: .6 });
  document.querySelectorAll("[data-to]").forEach(el => { fmt(el, +el.dataset.to); io.observe(el); });

  document.querySelector(".filters").addEventListener("click", e => {
    const b = e.target.closest("button"); if (!b) return;
    document.querySelectorAll(".filters button").forEach(x => x.setAttribute("aria-pressed", x === b));
    document.querySelectorAll(".role").forEach(r => r.hidden = b.dataset.team !== "all" && r.dataset.team !== b.dataset.team);
  });
