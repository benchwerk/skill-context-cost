  const jobs = [
    { title: "Senior Backend Engineer", team: "Engineering", loc: "Remote (EU/US)" },
    { title: "Security Engineer", team: "Engineering", loc: "London" },
    { title: "Staff Product Designer", team: "Product", loc: "New York" },
    { title: "Product Manager, Lending", team: "Product", loc: "Remote (US)" },
    { title: "Fraud Operations Analyst", team: "Operations", loc: "Lisbon" },
    { title: "Customer Success Lead", team: "Operations", loc: "Remote (EU)" }
  ];
  const list = document.getElementById("roles");
  const buttons = document.querySelectorAll(".filters button");
  function render(team) {
    list.innerHTML = "";
    jobs.filter(j => team === "all" || j.team === team).forEach(j => {
      const el = document.createElement("div");
      el.className = "role";
      el.innerHTML = `<div><h3></h3><small></small></div><a href="#">Apply →</a>`;
      el.querySelector("h3").textContent = j.title;
      el.querySelector("small").textContent = `${j.team} · ${j.loc}`;
      list.appendChild(el);
    });
  }
  buttons.forEach(b => b.addEventListener("click", () => {
    buttons.forEach(x => x.setAttribute("aria-pressed", x === b));
    render(b.dataset.team);
  }));
  render("all");

  // Count-up animation for key numbers (skipped when reduced motion is preferred)
  if (!matchMedia("(prefers-reduced-motion: reduce)").matches) {
    const io = new IntersectionObserver(entries => entries.forEach(e => {
      if (!e.isIntersecting) return;
      io.unobserve(e.target);
      const el = e.target, end = parseFloat(el.dataset.count), dec = +el.dataset.dec || 0;
      const pre = el.dataset.prefix || "", suf = el.dataset.suffix || "", t0 = performance.now();
      (function tick(t) {
        const p = Math.min((t - t0) / 1200, 1), v = end * (1 - Math.pow(1 - p, 3));
        el.textContent = pre + v.toLocaleString("en-US", { minimumFractionDigits: dec, maximumFractionDigits: dec }) + suf;
        if (p < 1) requestAnimationFrame(tick);
      })(t0);
    }), { threshold: .6 });
    document.querySelectorAll("[data-count]").forEach(el => io.observe(el));
  }
</script>
