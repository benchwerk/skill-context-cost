  const views = [
    { name:"Front", bg:"#ff7a3d", fill:"#2b303b" },

    { name:"Retro", bg:"#e8dcc2", fill:"#bfb08a" }

  const stage = document.getElementById("stage"), thumbs = document.getElementById("thumbs");
  function board(v) {
    let keys = "";
    for (let r = 0; r < 5; r++) for (let c = 0; c < 15; c++)
      keys += `<rect x="${24 + c * 24}" y="${30 + r * 24}" width="20" height="20" rx="4" fill="${v.bg}" opacity="${r % 2 ? .75 : 1}"/>`;
    return `<svg viewBox="0 0 400 180"><rect x="8" y="10" width="384" height="150" rx="14" fill="${v.fill}"/>${keys}</svg>`;
  }
  function show(i) {
    stage.innerHTML = board(views[i]);
    stage.setAttribute("aria-label", views[i].name + " view");
    [...thumbs.children].forEach((b, j) => b.setAttribute("aria-pressed", j === i));
  }
  views.forEach((v, i) => {
    const b = document.createElement("button");
    b.setAttribute("aria-label", v.name + " view");
    b.innerHTML = `<i style="background:${v.fill}; border-bottom:6px solid ${v.bg}"></i>`;
    b.onclick = () => show(i);
    thumbs.append(b);
  });
  show(0);

  const price = document.getElementById("price"), qty = document.getElementById("qty"), msg = document.getElementById("msg");
  const current = () => document.querySelector("input[name=sw]:checked");
