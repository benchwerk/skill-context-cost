<script>
const $ = id => document.getElementById(id);
const state = { step:1, email:"", pw:"", lang:null, goal:null };
const langs = [["🇪🇸","Spanish"],["🇫🇷","French"],["🇯🇵","Japanese"],["🇩🇪","German"],["🇰🇷","Korean"],["🇮🇹","Italian"]];
const goals = [["☕","Casual","5 min / day"],["📖","Regular","10 min / day"],["💪","Serious","15 min / day"],["🔥","Intense","30 min / day"]];

function choices(el, list, key, cls) {
  list.forEach(([e, name, sub]) => {
    const b = document.createElement("button");
    b.className = "opt " + cls; b.setAttribute("aria-pressed", "false");
    b.innerHTML = sub ? `<span><b>${name}</b></span><span>${sub}</span>` : `<span class="e">${e}</span><b>${name}</b>`;
    if (sub) b.innerHTML = `<span class="e">${e}</span><span style="flex:1"><b>${name}</b></span><small>${sub}</small>`;
    b.onclick = () => {
      state[key] = name;
      [...el.children].forEach(c => c.setAttribute("aria-pressed", c === b));
      validate();
    };
    el.appendChild(b);
  });
}
choices($("langs"), langs, "lang", "");
choices($("goals"), goals, "goal", "goal");

function validate() {
  state.email = $("email").value.trim(); state.pw = $("pw").value;
  const okEmail = /^\S+@\S+\.\S+$/.test(state.email), okPw = state.pw.length >= 8;
  const ok = [null, okEmail && okPw, !!state.lang, !!state.goal][state.step];
  $("next").disabled = !ok;
  return { okEmail, okPw };
}
function flag(input, msgEl, ok, text) {
  input.classList.toggle("bad", !ok && input.value !== "");
  msgEl.textContent = !ok && input.value !== "" ? text : "";
}
["email", "pw"].forEach(id => $(id).addEventListener("input", () => validate()));
$("email").addEventListener("blur", () => flag($("email"), $("emailMsg"), validate().okEmail, "Enter a valid email address."));
$("pw").addEventListener("blur", () => flag($("pw"), $("pwMsg"), validate().okPw, "Use at least 8 characters."));

function show(n) {
  state.step = n;
  document.querySelectorAll(".step").forEach((s, i) => s.classList.toggle("on", i + 1 === n));
  const t = Math.min(n, 3);
  $("fill").style.width = (n === 4 ? 100 : t * 33.3) + "%";
  document.querySelector(".bar").setAttribute("aria-valuenow", t);
  $("back").hidden = n === 1 || n === 4;
  $("next").textContent = n === 3 ? "Start learning" : "Continue";
  $("next").style.display = n === 4 ? "none" : "";
  if (n === 4) $("summary").textContent = `${state.lang} · ${state.goal} goal. Your first lesson is ready.`;
  validate();
}
$("next").onclick = () => { if (!$("next").disabled) show(state.step + 1); };
$("back").onclick = () => show(state.step - 1);
document.addEventListener("keydown", e => { if (e.key === "Enter" && state.step === 1) $("next").click(); });
show(1);
</script>
