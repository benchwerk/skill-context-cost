<script>
const $=s=>document.querySelector(s),all=s=>[...document.querySelectorAll(s)];
const langs=[["Spanish","Hola"],["Japanese","こんにちは"],["Portuguese","Olá"],["Korean","안녕하세요"],["Italian","Ciao"],["Turkish","Merhaba"]];
const goals=[[5,"Casual","One short lesson"],[10,"Regular","Two lessons"],[15,"Serious","Lessons plus review"],[20,"Intense","A full practice block"]];
$("#langs").innerHTML=langs.map(([l,h],i)=>`<label class="opt"><span><b>${l}</b><br><small>${h}</small></span><input type="radio" name="lang" value="${l}"${i?"":" checked"}></label>`).join("");
$("#goals").innerHTML=goals.map(([m,t,d],i)=>`<label class="opt"><span><b>${t}</b><br><small>${d}</small></span><span class="mins">${m} min</span><input type="radio" name="goal" value="${m}"${i==1?" checked":""}></label>`).join("");
let at=0;const f=$("#f"),bars=all(".progress li");
function show(n){at=n;all(".stage").forEach(s=>s.hidden=+s.dataset.s!==n);bars.forEach((b,i)=>b.classList.toggle("on",i<=Math.min(n,2)));$(`[data-s="${n}"] h1`).scrollIntoView({block:"start"});}
function bad(id,msg){const i=$("#"+id),e=$("#"+id+"-e");e.textContent=msg;msg?i.setAttribute("aria-invalid","true"):i.removeAttribute("aria-invalid");return !!msg;}
function checkAccount(){
  const em=/^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  const a=bad("n",$("#n").value.trim()?"":"Enter your first name.");
  const b=bad("e",em.test($("#e").value)?"":"Enter a valid email, like name@example.com.");
  const c=bad("p",$("#p").value.length>=8?"":"Use at least 8 characters.");
  if(a)$("#n").focus();else if(b)$("#e").focus();else if(c)$("#p").focus();
  return !(a||b||c);
}
f.addEventListener("submit",e=>{
  e.preventDefault();
  if(at===0){
    if(!checkAccount())return;
    f.classList.add("busy");
    setTimeout(()=>{f.classList.remove("busy");show(1)},800);
  }else if(at===1){
    if(!f.lang.value){$("#l-e").textContent="Choose a language to continue.";return}
    show(2);
  }else if(at===2){
    const g=goals.find(x=>x[0]==f.goal.value);
    $("#d-h").textContent=`You're in, ${$("#n").value.trim()}.`;
    $("#d-s").innerHTML=`<b>${f.lang.value}</b><span>${g[0]} minutes a day, ${g[1].toLowerCase()} pace</span>`;
    show(3);
  }
});
all("[data-back]").forEach(b=>b.addEventListener("click",()=>show(at-1)));
["n","e","p"].forEach(id=>$("#"+id).addEventListener("input",()=>bad(id,"")));
</script>
