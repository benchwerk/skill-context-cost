<script>
  document.documentElement.classList.add('js');
  const io = new IntersectionObserver(es => es.forEach(e => e.isIntersecting && (e.target.classList.add('in'), io.unobserve(e.target))), { threshold: 0.15 });
  document.querySelectorAll('.rv').forEach(el => io.observe(el));

  const steps = [...document.querySelectorAll('#steps li')], status = document.getElementById('status'), next = document.getElementById('next');
  const msg = ['Roasted this morning and resting.', 'Packed in a valve bag with your roast date.', 'On the road with the courier.', 'Delivered. Enjoy the first cup.'];
  let at = -1;
  function paint() {
    steps.forEach((s, i) => { s.className = i < at ? 'done' : i === at ? 'now' : ''; });
    status.textContent = at < 0 ? 'Nothing in transit yet. Press the button to start a demo order.' : msg[at];
    next.textContent = at === 3 ? 'Replay demo' : 'Advance demo';
  }
  next.onclick = () => { at = at === 3 ? -1 : at + 1; paint(); };
  document.getElementById('roast').onchange = () => { at = -1; paint(); };
  paint();

  document.getElementById('f').onsubmit = e => {
    e.preventDefault();
    const v = document.getElementById('em').value.trim(), err = document.getElementById('err');
    err.textContent = /^\S+@\S+\.\S+$/.test(v) ? '' : 'Enter a valid email address, like name@example.com.';
    if (!err.textContent) { err.style.color = 'var(--fg)'; err.textContent = 'Sent. Check your inbox for the download link.'; }
    else err.style.color = '';
  };
</script>
