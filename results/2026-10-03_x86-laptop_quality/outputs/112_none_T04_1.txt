    { t: "12:00", name: "Priya Raman",     type: "Whitening consult",   dr: "Dr. Patel", s: "wait" },
    { t: "13:30", name: "Tom Becker",      type: "Extraction",          dr: "Dr. Lee",   s: "wait" },
    { t: "14:15", name: "Lena Fischer",    type: "Check-up",            dr: "Dr. Patel", s: "late" },
    { t: "15:00", name: "Carlos Mendes",   type: "Braces adjustment",   dr: "Dr. Lee",   s: "cancel" },
    { t: "16:00", name: "Hannah Wright",   type: "Cleaning",            dr: "Dr. Patel", s: "wait" },
  ];
  const label = { done: "✓ Completed", now: "● In chair", wait: "○ Scheduled", late: "▲ Running late", cancel: "✕ Cancelled" };

  document.getElementById("schedule").innerHTML = appts.map(a => `
    <div class="appt ${a.s === "now" ? "next" : ""}">
      <div class="time">${a.t}</div>
      <div class="who"><b>${a.name}</b><span>${a.type} · ${a.dr}</span></div>
      <div class="tag ${a.s}">${label[a.s]}</div>
    </div>`).join("");

  const count = s => appts.filter(a => a.s === s).length;
  const active = appts.length - count("cancel");
  const stats = [
    ["Booked today", active, `${count("cancel")} cancelled`],
    ["Completed", count("done"), `of ${active} remaining on track`],
    ["Still to come", count("wait") + count("late"), "after current patient"],
    ["Chair utilisation", Math.round((active / 10) * 100) + "%", "10 slots available"],
  ];
  document.getElementById("stats").innerHTML = stats.map(([l, v, sub]) =>
    `<div class="card stat"><div class="l">${l}</div><div class="v">${v}</div><div class="l">${sub}</div></div>`).join("");

  const week = [
    { d: "Mon", n: 8, open: 2 }, { d: "Tue", n: 10, open: 0 }, { d: "Wed", n: 7, open: 3 },

      <div class="n">${w.n}</div>
      <div class="bar" style="height:${(w.n / max) * 100}%"></div>
    </div>`).join("");

  };

    c.addEventListener("mousemove", e => show(c, e.clientX, e.clientY));
    c.addEventListener("focus", () => { const r = c.getBoundingClientRect(); show(c, r.left, r.top); });
    c.addEventListener("mouseleave", () => tip.style.opacity = 0);
    c.addEventListener("blur", () => tip.style.opacity = 0);
  });
</script>
