<script>
  const steps = [...document.querySelectorAll('#steps li')];
  const eta = document.getElementById('eta'), sub = document.getElementById('sub'), fill = document.getElementById('fill');
  const copy = [
    ['Confirmed', 'We have your order, roasting next'],
    ['Roasting', 'In the drum, first crack at 7:40 am'],
    ['Resting', 'Degassing for peak flavor, ships tomorrow'],
    ['Arriving 4:10 pm', 'Courier is 6 stops away'],
    ['Delivered', 'Left at your front door. Enjoy.']
  ];
  let i = 1;
  function show(n) {
    steps.forEach((li, k) => { li.className = k < n ? 'done' : k === n ? 'on' : ''; });
    eta.textContent = copy[n][0]; sub.textContent = copy[n][1];
    fill.style.width = ((n + 1) / steps.length * 100) + '%';
  }
  show(i);
  if (!matchMedia('(prefers-reduced-motion: reduce)').matches)
    setInterval(() => { i = (i + 1) % steps.length; show(i); }, 2600);
</script>
