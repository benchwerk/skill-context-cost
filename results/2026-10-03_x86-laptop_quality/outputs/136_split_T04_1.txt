<script>
const SLOTS = 12; // bookable chair slots per day
const appts = [
  {t:"08:30", n:"Maren Okafor",    p:"Cleaning and exam",   s:"done"},
  {t:"09:15", n:"Daniel Whitfield", p:"Crown fitting",       s:"done"},
  {t:"10:00", n:"Priya Raman",      p:"Root canal, step 2",  s:"in"},
  {t:"11:00", n:"Tomas Eriksen",    p:"Filling, lower left", s:"late"},
  {t:"11:45", n:"Hana Yoshida",     p:"Whitening consult",   s:"wait"},
  {t:"13:30", n:"Leo Castellanos",  p:"Child checkup",       s:"wait"},
  {t:"14:15", n:"Ruth Adeyemi",     p:"Extraction review",   s:"wait"},
  {t:"15:30", n:"Samir Haddad",     p:"Cleaning and exam",   s:"wait"}
];
const week = [["Mon",11],["Tue",8],["Wed",9],["Thu",11],["Fri",5]];
const labels = {done:"Seen", in:"In chair", late:"Running late", wait:"Waiting"};
const order = ["wait","in","done"];
const $ = id => document.getElementById(id);

$("date").textContent = new Date().toLocaleDateString(undefined,{weekday:"long",month:"long",day:"numeric"}) + " · Dr. Alvarez, Dr. Lim";

function render(){
  $("list").innerHTML = appts.map((a,i) => `
    <div class="row ${a.s}" tabindex="0" role="button" data-i="${i}" aria-label="${a.n}, ${labels[a.s]}">
      <span class="time">${a.t}</span>
      <div class="who"><b>${a.n}</b><span>${a.p}</span></div>
      <span class="tag">${labels[a.s]}</span>
    </div>`).join("");
  const done = appts.filter(a => a.s === "done").length;
  $("s-total").textContent = appts.length;
  $("s-left").textContent = (appts.length - done) + " still to come";
  $("s-done").textContent = done;
  $("s-use").textContent = Math.round(appts.length / SLOTS * 100) + "%";
  $("s-open").textContent = SLOTS - appts.length;
}

function advance(i){
  const a = appts[i];
  a.s = order[Math.min(order.indexOf(a.s === "late" ? "wait" : a.s) + 1, 2)];
  render();
}

$("list").addEventListener("click", e => { const r = e.target.closest(".row"); if (r) advance(+r.dataset.i); });
$("list").addEventListener("keydown", e => {
  if (e.key !== "Enter" && e.key !== " ") return;
  const r = e.target.closest(".row"); if (!r) return;
  e.preventDefault(); const i = +r.dataset.i; advance(i);
  $("list").querySelector(`[data-i="${i}"]`).focus();
});
$("next").addEventListener("click", () => {
  const i = appts.findIndex(a => a.s === "wait" || a.s === "late");
  if (i > -1) { appts[i].s = "in"; render(); }
});

const today = (new Date().getDay() + 6) % 7; // Mon = 0
$("week").innerHTML = week.map(([d,c],i) => `
  <div class="day ${i === Math.min(today,4) ? "today" : ""}">
    <div class="bar" style="height:${c / SLOTS * 100}%;animation-delay:${i * 60}ms"></div>
    <b>${c}</b><span>${d}</span>
  </div>`).join("");

render();
</script>
