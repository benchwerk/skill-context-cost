<script>
  const colors = { Slate:["#3d4452","#59627a"], Sand:["#cbb99a","#e6dac1"], Coral:["#e4572e","#f08a68"] };
  const views = ["Top", "Angle", "Side"];
  const reviews = [
    { n:"Maya R.", r:5, t:"Typing feels buttery. The gasket mount sounds incredible out of the box." },
    { n:"Dev P.", r:4, t:"Great build and battery life. Wish the knob were programmable without software." },
    { n:"Lena K.", r:4, t:"Hot-swap made trying switches painless. Bluetooth reconnects quickly." }
  ];
  const $ = s => document.querySelector(s);
  let view = 0;

  function board(v, c) {
    const [body, key] = colors[c];
    const skew = v === 1 ? 'transform="skewX(-14) translate(40 0)"' : '';
    const h = v === 2 ? 40 : 150;
    let keys = "";
    if (v !== 2) for (let r = 0; r < 5; r++) for (let i = 0; i < 14; i++)
      keys += `<rect x="${38 + i * 24}" y="${34 + r * 22}" width="20" height="18" rx="4" fill="${key}"/>`;
    return `<svg viewBox="0 0 400 ${v === 2 ? 90 : 190}" role="img" aria-label="${views[v]} view">
      <g ${skew}><rect x="22" y="${v === 2 ? 20 : 18}" width="356" height="${h}" rx="14" fill="${body}"/>${keys}</g></svg>`;
  }
  function render() {
    const c = document.querySelector('[name=col]:checked').value;
    $("#stage").innerHTML = board(view, c);
    $("#thumbs").innerHTML = views.map((n, i) =>
      `<button aria-pressed="${i === view}" aria-label="${n} view" data-i="${i}">${board(i, c)}</button>`).join("");
  }
  $("#thumbs").onclick = e => {
    const b = e.target.closest("button");
    if (b) { view = +b.dataset.i; render(); }
  };
  document.querySelectorAll("input").forEach(i => i.onchange = () => {
    $("#price").textContent = "$" + document.querySelector("[name=sw]:checked").dataset.p;
    render();
  });
  $("#buy").onclick = () => {
    const sw = document.querySelector("[name=sw]:checked").value, c = document.querySelector("[name=col]:checked").value;
    $("#msg").textContent = `Added: Halcyon 75, ${c}, ${sw} switches — ${$("#price").textContent}`;
  };
  $("#reviews").innerHTML = reviews.map(r => `<div class="review"><header><span>${r.n}</span>
    <span class="stars">${"★".repeat(r.r)}${"☆".repeat(5 - r.r)}</span></header><p>${r.t}</p></div>`).join("");
  render();
</script>
