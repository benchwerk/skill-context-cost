<script>
  const langs = [["🇪🇸","Spanish","Most popular"],["🇫🇷","French","Language of love"],["🇯🇵","Japanese","Kana to kanji"],["🇩🇪","German","Der, die, das"],["🇮🇹","Italian","Bellissimo"]];
  const goals = [["☕","Casual","5 min / day"],["📖","Regular","10 min / day"],["💪","Serious","20 min / day"],["🚀","Intense","30 min / day"]];
  const state = { step:1, lang:null, goal:null };
  const $ = id => document.getElementById(id);

  function build(el, items, key) {
    items.forEach(([em, name, sub]) => {
      const b = document.createElement("button");
      b.className = "opt"; b.setAttribute("aria-pressed", "false");
      b.innerHTML = `<span class="em">${em}</span><span><b>${name}</b><small>${sub}</small></span>`;
      b.onclick = () => {
        state[key] = name;
        [...el.children].forEach(c => c.setAttribute("aria-pressed", c === b));
        update();
      };
      el.appendChild(b);
    });
  }
  build($("langs"), langs, "lang");
  build($("goals"), goals, "goal");

  function validEmail() { return /^\S+@\S+\.\S+$/.test($("email").value.trim()); }
  function validPw() { return $("pw").value.length >= 8; }

  function update() {
    document.querySelectorAll(".step").forEach((s, i) => s.classList.toggle("active", i + 1 === state.step));
    [...$("bar").children].forEach((b, i) => b.classList.toggle("on", i < Math.min(state.step, 3)));
    $("bar").setAttribute("aria-valuenow", Math.min(state.step, 3));
    $("back").classList.toggle("show", state.step > 1 && state.step < 4);
    $("next").style.display = state.step === 4 ? "none" : "";
    $("next").textContent = state.step === 3 ? "Start learning" : "Continue";
    $("next").disabled = (state.step === 1 && !(validEmail() && validPw())) ||
                         (state.step === 2 && !state.lang) || (state.step === 3 && !state.goal);
  }

  ["email", "pw"].forEach(id => {
    $(id).addEventListener("input", () => { $(id).classList.remove("bad"); $("err1").textContent = ""; update(); });
    $(id).addEventListener("blur", () => {
      if (!$(id).value) return;
      const ok = id === "email" ? validEmail() : validPw();
      $(id).classList.toggle("bad", !ok);
      $("err1").textContent = ok ? "" : id === "email" ? "Enter a valid email address." : "Password must be at least 8 characters.";
    });
  });

  $("next").onclick = () => {
    if (state.step === 3) $("summary").textContent = `${state.lang} · ${state.goal} goal. Your first lesson is ready.`;
    state.step++; update();
  };
  $("back").onclick = () => { state.step--; update(); };
  update();
</script>
