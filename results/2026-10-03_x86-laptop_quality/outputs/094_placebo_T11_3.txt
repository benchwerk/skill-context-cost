<script>
  const P = [
    ["Switchback 2P", "Tent", 389, 1.6, 150], ["Cairn 1P", "Tent", 299, 1.1, 200], ["Basecamp 4P", "Tent", 520, 3.9, 30],
    ["Ember 15° Down", "Sleeping bag", 340, 0.9, 60], ["Drift 30° Synthetic", "Sleeping bag", 190, 1.3, 90], ["Summit 0° Down", "Sleeping bag", 480, 1.4, 20],
    ["AirLite Pad", "Pad", 149, 0.4, 120], ["Terrain Self-Inflate", "Pad", 99, 0.9, 160], ["Foam Z-Fold", "Pad", 45, 0.4, 240],
    ["Silcloth Tarp 3x3", "Tarp", 120, 0.5, 280], ["Ridge Bivy", "Tarp", 210, 0.7, 330], ["Hammock Kit", "Tarp", 175, 0.8, 100]
  ].map(([name, type, price, kg, h], i) => ({ name, type, price, kg, h, i }));
  const $ = id => document.getElementById(id);
  const types = [...new Set(P.map(p => p.type))];
  const state = { types: new Set(types), kg: 4, sort: "featured" }, cart = { n: 0 };
  $("types").innerHTML = types.map(t =>
    `<label><span><input type="checkbox" value="${t}" checked>${t}</span><em>${P.filter(p => p.type === t).length}</em></label>`).join("");
  function render() {
    const list = P.filter(p => state.types.has(p.type) && p.kg <= state.kg)
      .sort((a, b) => state.sort === "featured" ? a.i - b.i : a[state.sort] - b[state.sort]);
    $("wOut").textContent = state.kg.toFixed(1) + " kg";
    $("count").textContent = list.length + " of 12 products";
    $("grid").innerHTML = list.length ? list.map(p => `
      <li class="card"><div class="art" style="--h:${p.h}"><b>${p.type}</b></div>
        <div class="body"><h3>${p.name}</h3><p class="meta">${p.kg.toFixed(1)} kg</p>
          <div class="row"><span class="price">$${p.price}</span><button class="add" data-n="${p.name}">Add</button></div></div></li>`).join("")
      : `<li class="empty">Nothing fits that weight. Loosen the slider or clear filters.</li>`;
  }
  $("types").onchange = e => { e.target.checked ? state.types.add(e.target.value) : state.types.delete(e.target.value); render(); };
  $("weight").oninput = e => { state.kg = +e.target.value; render(); };
  $("sort").onchange = e => { state.sort = e.target.value; render(); };
  $("grid").onclick = e => { if (e.target.matches(".add")) { $("cart").textContent = "Cart: " + ++cart.n; e.target.textContent = "Added"; } };
  $("reset").onclick = () => {
    state.types = new Set(types); state.kg = 4; state.sort = "featured";
    $("weight").value = 4; $("sort").value = "featured";
    document.querySelectorAll("#types input").forEach(c => c.checked = true); render();
  };
  render();
</script>
