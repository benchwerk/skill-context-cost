const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) { e.target.classList.add('in'); io.unobserve(e.target); }
}), { threshold: .15 });
document.querySelectorAll('.rv').forEach((el, i) => { el.style.transitionDelay = (i % 3) * 90 + 'ms'; io.observe(el); });

const date = document.getElementById('date');
date.min = new Date().toISOString().split('T')[0];

document.getElementById('rf').addEventListener('submit', e => {
  e.preventDefault();
  const f = new FormData(e.target);
  if (new Date(f.get('date') + 'T12:00').getDay() === 1) {
    document.getElementById('note').textContent = 'We are closed on Mondays. Please choose another day.';
    return;
  }
  document.getElementById('note').textContent =
    `Thank you, ${f.get('name')}. We have noted a table for ${f.get('guests')} on ${f.get('date')} at ${f.get('time')} and will confirm by phone.`;
  e.target.reset();
});
