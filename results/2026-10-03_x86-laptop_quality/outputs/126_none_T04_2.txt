<script>
const STATUS = {
  done:  { cls:"s-done",  icon:"✓", label:"Completed" },
  chair: { cls:"s-chair", icon:"●", label:"In chair" },
  wait:  { cls:"s-wait",  icon:"!", label:"Checked in" },
  conf:  { cls:"s-conf",  icon:"○", label:"Confirmed" }
};
const appts = [
  ["08:30","Amara Okafor","Check-up & cleaning","Dr. Lee","done"],
  ["09:15","Tom Brandt","Filling (lower left)","Dr. Patel","done"],
  ["10:00","Sofia Marquez","Root canal, visit 2","Dr. Lee","chair"],
  ["10:45","Henrik Olsen","Crown fitting","Dr. Patel","wait"],
  ["11:30","Priya Raman","Whitening consult","Dr. Lee","conf"],
  ["13:30","Daniel Kim","Child exam","Dr. Patel","conf"],
  ["14:15","Linda Walsh","Extraction","Dr. Lee","conf"],
  ["15:30","Marcus Bell","Scale & polish","Hygienist","conf"]
];
const week = [["Mon",11],["Tue",9],["Wed",8],["Thu",10],["Fri",7],["Sat",4]];
const todayIdx = 2;

document.getElementById("date").textContent =
  new Date().toLocaleDateString(undefined,{weekday:"long",day:"numeric",month:"long",year:"numeric"});

const done = appts.filter(a => a[4] === "done").length;
const weekTotal = week.reduce((s,d) => s + d[1], 0);
const stats = [
  ["Today's appointments", appts.length, `${done} completed, ${appts.length - done} remaining`],
  ["Confirmed", Math.round(appts.filter(a => a[4] !== "wait").length / appts.length * 100) + "%", "of today's bookings"],
  ["This week", weekTotal, "appointments booked"],
  ["Open slots today", 3, "next: 12:15 and 16:15"]
];
document.getElementById("stats").innerHTML = stats.map(s =>
  `<div class="card stat"><div class="label">${s[0]}</div><div class="value">${s[1]}</div><div class="note">${s[2]}</div></div>`).join("");

document.getElementById("rows").innerHTML = appts.map(a => {
  const s = STATUS[a[4]];
  return `<tr class="${a[4] === "chair" ? "now" : ""}"><td class="time">${a[0]}</td><td class="patient">${a[1]}</td>
    <td class="muted">${a[2]}</td><td class="muted">${a[3]}</td>
    <td><span class="status ${s.cls}"><i aria-hidden="true">${s.icon}</i>${s.label}</span></td></tr>`;
}).join("");

const max = Math.max(...week.map(d => d[1]));
document.getElementById("chart").innerHTML = week.map((d,i) =>
  `<div class="col ${i === todayIdx ? "today" : ""}" tabindex="0" data-tip="${d[0]}: ${d[1]} appointments">
     <div class="n">${d[1]}</div><div class="bar" style="height:${d[1] / max * 85}%"></div></div>`).join("");
document.getElementById("days").innerHTML = week.map((d,i) =>
  `<span class="${i === todayIdx ? "today" : ""}">${d[0]}</span>`).join("");

const tip = document.getElementById("tip");
document.querySelectorAll(".col").forEach(c => {
  const show = e => {
    const r = c.getBoundingClientRect();
    tip.textContent = c.dataset.tip;
    tip.style.left = (r.left + r.width / 2 - tip.offsetWidth / 2) + "px";
    tip.style.top = (r.top - 4) + "px";
    tip.style.opacity = 1;
  };
  c.addEventListener("mouseenter", show);
  c.addEventListener("focus", show);
  c.addEventListener("mouseleave", () => tip.style.opacity = 0);
  c.addEventListener("blur", () => tip.style.opacity = 0);
});
</script>
