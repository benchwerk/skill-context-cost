<script>
const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
const fmt = (el, v) => {
  const d = +el.dataset.dec || 0;
  el.textContent = (el.dataset.pre || '') + v.toLocaleString('en-US', {minimumFractionDigits: d, maximumFractionDigits: d}) + (el.dataset.suf || '');
};
const io = new IntersectionObserver(entries => entries.forEach(e => {
  if (!e.isIntersecting) return;
  io.unobserve(e.target);
  const el = e.target, to = +el.dataset.to, t0 = performance.now();
  const tick = now => {
    const p = Math.min((now - t0) / 1200, 1);
    fmt(el, to * (1 - Math.pow(1 - p, 3)));
    if (p < 1) requestAnimationFrame(tick);
  };
  requestAnimationFrame(tick);
}), {threshold: .6});
if (!reduce) document.querySelectorAll('[data-to]').forEach(el => io.observe(el));

const buttons = document.querySelectorAll('.filters button');
buttons.forEach(b => b.addEventListener('click', () => {
  buttons.forEach(x => x.setAttribute('aria-pressed', x === b));
  document.querySelectorAll('.roles li').forEach(li => li.hidden = b.dataset.t !== 'all' && li.dataset.t !== b.dataset.t);
}));
</script>
