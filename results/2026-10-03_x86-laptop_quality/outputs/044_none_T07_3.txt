<script>
  const f = document.getElementById('f'), next = document.getElementById('next'), back = document.getElementById('back');
  const secs = [...f.querySelectorAll('section')], bars = document.querySelectorAll('.steps i');
  const bar = document.querySelector('.steps');
  let step = 1;

  function show(n) {
    step = n;
    secs.forEach(s => s.classList.toggle('active', +s.dataset.step === n));
    bars.forEach((b, i) => b.classList.toggle('on', i < Math.min(n, 3)));
    bar.setAttribute('aria-valuenow', Math.min(n, 3));
    back.hidden = n === 1 || n === 4;
    next.textContent = n === 3 ? 'Start learning' : 'Continue';
    next.hidden = n === 4;
    const focus = secs[n - 1].querySelector('input');
    if (focus) focus.focus({ preventScroll: true });
  }

  function validate() {
    let ok = true;
    secs[0].querySelectorAll('input').forEach(inp => {
      const msg = inp.validity.valueMissing ? 'This field is required.'
        : inp.validity.typeMismatch ? 'Enter a valid email address.'
        : inp.validity.tooShort ? 'Use at least 8 characters.' : '';
      inp.setAttribute('aria-invalid', !!msg);
      inp.parentElement.querySelector('.err').textContent = msg;
      if (msg && ok) { inp.focus(); ok = false; }
    });
    return ok;
  }

  f.addEventListener('submit', e => {
    e.preventDefault();
    if (step === 1 && !validate()) return;
    if (step < 3) return show(step + 1);
    const d = new FormData(f);
    document.getElementById('doneTitle').textContent = `Welcome, ${d.get('name').trim().split(' ')[0]}!`;
    document.getElementById('doneText').textContent = `${d.get('goal')} minutes of ${d.get('lang')} a day. Your first lesson is ready.`;
    show(4);
  });
  back.addEventListener('click', () => show(step - 1));
</script>
