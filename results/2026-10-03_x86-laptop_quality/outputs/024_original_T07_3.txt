<script>
  const form = document.querySelector("form");
  const panels = [...form.querySelectorAll(".panel")];
  const steps = [...document.querySelectorAll(".progress li")];
  const $ = (id) => document.getElementById(id);
  let stage = 0;

  function show(n) {
    stage = n;
    panels.forEach((p, i) => (p.hidden = i !== n));
    steps.forEach((s, i) => {
      s.classList.toggle("on", i === Math.min(n, 2));
      s.classList.toggle("done", i < n);
      i === Math.min(n, 2) ? s.setAttribute("aria-current", "step") : s.removeAttribute("aria-current");
    });
    panels[n].querySelector("input, select, h2")?.focus?.({ preventScroll: false });
  }

  function fail(id, msg) {
    $(id).setAttribute("aria-invalid", "true");
    $(id + "-e").textContent = msg;
    return false;
  }

  function validateAccount() {
    ["name", "email", "pw"].forEach((id) => { $(id).removeAttribute("aria-invalid"); $(id + "-e").textContent = ""; });
    const ok = [
      $("name").value.trim() || fail("name", "Enter your first name."),
      /^\S+@\S+\.\S+$/.test($("email").value) || fail("email", "Enter an email like name@example.com."),
      $("pw").value.length >= 8 || fail("pw", "Use at least 8 characters."),
    ];
    if (!ok.every(Boolean)) document.querySelector("[aria-invalid]").focus();
    return ok.every(Boolean);
  }

  form.addEventListener("submit", (e) => {
    e.preventDefault();
    if (stage === 0 && !validateAccount()) return;
    if (stage < 2) return show(stage + 1);
    if (stage === 2) {
      const btn = e.submitter;
      btn.setAttribute("aria-busy", "true");
      btn.textContent = "Creating your plan...";
      setTimeout(() => {
        const goal = form.elements.goal.value;
        $("done-title").textContent = "You're in, " + $("name").value.trim() + ".";
        $("done-text").textContent = $("lang").value + ", " + goal + " a day. Your first lesson is ready.";
        show(3);
      }, 900);
    }
  });

  form.querySelectorAll("[data-back]").forEach((b) => b.addEventListener("click", () => show(stage - 1)));
</script>
