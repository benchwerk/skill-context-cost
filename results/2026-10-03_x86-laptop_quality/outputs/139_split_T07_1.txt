<script>
  const $ = (s, r = document) => r.querySelector(s), $$ = (s, r = document) => [...r.querySelectorAll(s)];
  const steps = $$('section'), bars = $$('.bar i');
  let step = 1;

  function go(n) {
    step = n;
    steps.forEach(s => s.hidden = +s.dataset.step !== n);
    bars.forEach((b, i) => b.classList.toggle('on', i < Math.min(n, 3)));
    $('#count').textContent = n < 4 ? `Step ${n} of 3` : 'Welcome';
    $('#back').hidden = n === 1 || n === 4;
    $('#next').textContent = n === 3 ? 'Start learning' : n === 4 ? 'Go to first lesson' : 'Continue';
    $('#fine').hidden = n !== 1;
    if (n === 3) updateTotal();
    $('h1', steps[n - 1]).scrollIntoView({ block: 'start' });
  }

  function updateTotal() {
    const m = +$('[name=goal]:checked').value;
    $('#total').innerHTML = `That is <strong>${Math.round(m * 7 / 6) * 6 / 6 * 1} min</strong> a week, about <strong>${(m * 30 / 60).toFixed(m * 30 % 60 ? 1 : 0)} hours</strong> a month.`;
  }

  function validate() {
    const email = $('#email'), pw = $('#pw');
    const msg = !/^\S+@\S+\.\S+$/.test(email.value) ? 'Enter a valid email address.'
              : pw.value.length < 8 ? 'Password needs at least 8 characters.' : '';
    $('#err').textContent = msg;
    if (msg) (email.validity.valid && /^\S+@\S+\.\S+$/.test(email.value) ? pw : email).focus();
    return !msg;
  }

  $('#flow').addEventListener('submit', e => {
    e.preventDefault();
    if (step === 1 && !validate()) return;
    if (step === 3) {
      $('#who').textContent = $('#email').value.split('@')[0];
      $('#summary').textContent = `${$('[name=lang]:checked').value}, ${$('[name=goal]:checked').value} minutes a day. See you tomorrow.`;
    }
    if (step < 4) go(step + 1);
  });
  $('#back').addEventListener('click', () => go(step - 1));
  $$('[name=goal]').forEach(r => r.addEventListener('change', updateTotal));
  $('#show').addEventListener('click', e => {
    const hide = $('#pw').type === 'password';
    $('#pw').type = hide ? 'text' : 'password';
    e.target.textContent = hide ? 'Hide' : 'Show';
    e.target.setAttribute('aria-pressed', hide);
  });
</script>
