<script>
  const sections = [...document.querySelectorAll("section")];
  const bars = [...document.querySelectorAll(".progress li")];
  const $ = id => document.getElementById(id);
  let step = 0;

  function show(n, focus = true) {
    step = n;
    sections.forEach((s, i) => s.classList.toggle("on", i === n));
    bars.forEach((b, i) => { b.className = i < n ? "done" : i === n ? "now" : ""; });
    const last = n === sections.length - 1;
    $("stepno").textContent = last ? "All set" : `Step ${n + 1} of 3 - ${sections[n].dataset.title}`;
    $("back").hidden = n === 0 || last;
    $("actions").hidden = $("fine").hidden = last;
    $("next").textContent = n === 2 ? "Start learning" : "Continue";
    $("fine").hidden = n !== 0;
    if (focus) sections[n].querySelector("h1").focus({ preventScroll: true });
    scrollTo({ top: 0 });
  }

  function validate() {
    if (step !== 0) return true;
    const email = $("email"), pw = $("pw");
    const okE = email.checkValidity() && email.value.trim() !== "";
    const okP = pw.value.length >= 8;
    $("e-email").textContent = okE ? "" : "Enter a valid email address.";
    $("e-pw").textContent = okP ? "" : "Use at least 8 characters.";
    email.setAttribute("aria-invalid", !okE); pw.setAttribute("aria-invalid", !okP);
    (okE ? pw : email).focus();
    return okE && okP;
  }

  $("flow").addEventListener("submit", e => {
    e.preventDefault();
    if (!validate()) return;
    if (step < 2) return show(step + 1);
    const lang = document.querySelector("[name=lang]:checked").value;
    const goal = document.querySelector("[name=goal]:checked").value;
    $("doneP").textContent = `${goal} minutes of ${lang} a day. Your first lesson is ready when you are.`;
    show(3);
  });
  $("back").addEventListener("click", () => show(step - 1));
  show(0, false);
</script>
