<script>
  const mo = document.getElementById('mo'), yr = document.getElementById('yr');
  function setBilling(yearly) {
    mo.setAttribute('aria-pressed', String(!yearly));
    yr.setAttribute('aria-pressed', String(yearly));
    document.querySelectorAll('.amt').forEach(el => {
      el.textContent = '$' + (yearly ? el.dataset.yr : el.dataset.mo);
    });
    document.querySelectorAll('.bill').forEach(el => {
      el.textContent = yearly ? 'Billed yearly' : 'Billed monthly';
    });
  }
  mo.onclick = () => setBilling(false);
  yr.onclick = () => setBilling(true);
</script>
