<script>
  const $ = id => document.getElementById(id);
  const colors = { Slate:"#4a5560", Sand:"#d8c9a8", Coral:"#e4572e" };
  const switches = { Linear:0, Tactile:10, Clicky:15 };
  let color = "Slate", sw = "Linear", view = 0;
  const keys = (c) => { let k = ""; for (let r = 0; r < 4; r++) for (let i = 0; i < 12; i++)
    k += `<rect x="${30 + i*30}" y="${60 + r*30}" width="26" height="26" rx="4" fill="#f4f1ea"/>`; return k; };
  const views = [
    c => `<rect x="14" y="40" width="392" height="148" rx="14" fill="${c}"/>${keys()}`,
    c => `<g transform="translate(0 40) scale(1 .6)"><rect x="14" y="40" width="392" height="148" rx="14" fill="${c}"/>${keys()}</g>`,
    c => `<circle cx="210" cy="120" r="70" fill="#f4f1ea"/><circle cx="210" cy="120" r="34" fill="${c}"/><text x="210" y="205" text-anchor="middle" font-size="14" fill="#6b6b72">Hot-swap switch</text>`
  ];
  const svg = (i, c) => `<svg viewBox="0 0 420 240" role="img" aria-label="View ${i+1}">${views[i](c)}</svg>`;
  function draw() {
    $("stage").innerHTML = svg(view, colors[color]);
    $("thumbs").innerHTML = "";
    views.forEach((_, i) => {
      const b = document.createElement("button");
      b.innerHTML = svg(i, colors[color]);
      b.setAttribute("aria-pressed", i === view);
      b.setAttribute("aria-label", "Show view " + (i + 1));
      b.onclick = () => { view = i; draw(); };
      $("thumbs").append(b);
    });
    $("price").textContent = "$" + (149 + switches[sw]);
  }
  function chips(id, opts, get, set) {
    const box = $(id); box.innerHTML = "";
    Object.keys(opts).forEach(n => {
      const b = document.createElement("button");
      b.textContent = n; b.setAttribute("aria-pressed", get() === n);
      b.onclick = () => { set(n); chips(id, opts, get, set); draw(); };
      box.append(b);
    });
  }
  const reviews = [
    { n:"Maya R.", r:5, t:"Sounds unreal out of the box. The gasket mount feels bouncy without being mushy." },
    { n:"Devon K.", r:4, t:"Great build and battery life. Wish the Bluetooth switched devices a bit faster." },
    { n:"Sam T.", r:4, t:"Hot-swap made trying switches painless. Keycaps feel premium for the price." }
  ];
  function renderReviews() {
    $("reviews").innerHTML = reviews.map(v => `<div class="review"><b>${v.n} <span class="stars">${"★".repeat(v.r)}${"☆".repeat(5 - v.r)}</span></b><small>Verified buyer</small><p>${v.t.replace(/[<>&]/g, "")}</p></div>`).join("");
    const avg = reviews.reduce((s, v) => s + v.r, 0) / reviews.length;
    $("avg").textContent = avg.toFixed(1) + " (" + reviews.length + " reviews)";
    $("count").textContent = avg.toFixed(1) + " out of 5 · " + reviews.length + " reviews";
  }
  $("form").onsubmit = e => {
    e.preventDefault();
    reviews.unshift({ n:$("name").value, r:+$("rating").value, t:$("text").value });
    e.target.reset(); renderReviews();
  };
  $("buy").onclick = () => {
    const q = Math.max(1, +$("qty").value || 1);
    $("msg").textContent = `Added ${q} × Halcyon 75 (${color}, ${sw}) – $${q * (149 + switches[sw])}`;
  };
  chips("switches", switches, () => sw, v => sw = v);
  chips("colors", colors, () => color, v => color = v);
  draw(); renderReviews();
</script>
