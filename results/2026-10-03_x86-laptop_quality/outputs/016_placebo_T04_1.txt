<script>
  const appts = [
    { t: "08:30", n: "Maya Okafor",     r: "Cleaning & exam",       m: 60, s: "done" },
    { t: "09:30", n: "Tomas Lindqvist", r: "Crown fitting",         m: 90, s: "done" },
    { t: "11:00", n: "Priya Raman",     r: "Root canal, follow-up", m: 45, s: "now" },
    { t: "11:45", n: "Elena Rossi",     r: "New patient exam",      m: 60, s: "next", fresh: true },
    { t: "13:30", n: "Daniel Whitaker", r: "Filling (lower left)",  m: 45, s: "next" },
    { t: "14:30", n: "Amara Nwosu",     r: "Whitening consult",     m: 30, s: "late" },
    { t: "15:15", n: "Jonas Becker",    r: "Night guard fitting",   m: 45, s: "next", fresh: true },
  ];
  const week = [8, 9, 6, 10, 7, 4]; // Mon–Sat bookings
  const names = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
  const todayIdx = (new Date().getDay() + 6) % 7;
  const label = { done: "Done", now: "In chair", next: "Upcoming", late: "Running late" };
  const $ = id => document.getElementById(id);

  $("date").textContent = new Date().toLocaleDateString("en-US", { weekday: "long", month: "long", day: "numeric" }) + " · Dr. Hale & Dr. Mensah";

  function render() {
    $("list").innerHTML = appts.map(a => `
      <li class="appt ${a.s === "done" ? "done" : ""}">
        <span class="time">${a.t}</span>
        <div class="who"><strong>${a.n}</strong><span>${a.r} · ${a.m} min${a.fresh ? " · New" : ""}</span></div>
        <span class="tag ${a.s === "now" || a.s === "late" ? a.s : ""}">${label[a.s]}</span>
      </li>`).join("");
    const left = appts.filter(a => a.s !== "done").length;
    const mins = appts.reduce((s, a) => s + a.m, 0);
    week[todayIdx] = Math.min(10, appts.length + 3);
    $("sTotal").textContent = appts.length;
    $("sLeft").textContent = left + " still to come";
    $("sHours").textContent = (mins / 60).toFixed(1).replace(".0", "") + " h";
    $("sNew").textContent = appts.filter(a => a.fresh).length;
    $("sOpen").textContent = week.reduce((s, n) => s + (10 - n), 0);
    $("count").textContent = appts.length + " patients";
    $("week").innerHTML = names.map((d, i) => `
      <button class="day ${i === todayIdx ? "today" : ""}" data-i="${i}" aria-label="${d}: ${week[i]} of 10 booked">
        <span>${d}</span><div class="bar"><i style="width:${week[i] * 10}%"></i></div><em>${week[i]}</em>
      </button>`).join("");
  }

  $("week").addEventListener("click", e => {
    const b = e.target.closest(".day");
    if (!b) return;
    const i = +b.dataset.i, free = 10 - week[i];
    $("note").textContent = `${names[i]}: ${week[i]} booked, ${free ? free + " slot" + (free > 1 ? "s" : "") + " open" : "fully booked"}.`;
  });

  function addWalkIn() {
    const name = prompt("Walk-in patient name?");
    if (!name || !name.trim()) return;
    appts.push({ t: "16:00", n: name.trim(), r: "Walk-in", m: 30, s: "next" });
    render();
  }
  render();
</script>
