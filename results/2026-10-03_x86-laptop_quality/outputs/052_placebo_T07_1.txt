<script>
  const form = document.getElementById('flow');
  const steps = [...form.querySelectorAll('.step')];
  const bars = document.querySelectorAll('.progress i');
  const back = document.getElementById('back');
  const names = ['Account', 'Language', 'Daily goal'];
  let cur = 0;

  function show(n) {
    cur = n;
    steps.forEach((s, i) => s.classList.toggle('active', i === n));
    bars.forEach((b, i) => b.classList.toggle('on', i <= n));
    document.getElementById('count').textContent = `Step ${n + 1} of 3 · ${names[n]}`;
    back.hidden = n === 0;
    steps[n].querySelector('input:not([type=radio]), input:checked')?.focus();
  }

  function validate() {
    const err = steps[0].querySelector('.err');
    const f = form.elements;
    const checks = [
      [f.name, f.name.value.trim().length > 0, 'Tell us what to call you.'],
      [f.email, /^\S+@\S+\.\S+$/.test(f.email.value), 'Enter a valid email address.'],
      [f.pw, f.pw.value.length >= 8, 'Password needs at least 8 characters.'],
    ];
    checks.forEach(([el, ok]) => el.setAttribute('aria-invalid', !ok));
    const bad = checks.find(c => !c[1]);
    err.textContent = bad ? bad[2] : '';
    if (bad) bad[0].focus();
    return !bad;
  }

  form.addEventListener('submit', e => {
    e.preventDefault();
    if (cur === 0 && !validate()) return;
    if (cur < 2) return show(cur + 1);
    const d = Object.fromEntries(new FormData(form));
    form.hidden = true;
    document.querySelector('.progress').hidden = true;
    document.getElementById('count').hidden = true;
    back.hidden = true;
    document.getElementById('done').hidden = false;
    document.getElementById('doneTitle').textContent = `Welcome, ${d.name.trim()}!`;
    document.getElementById('doneText').textContent =
      `${d.goal} minutes of ${d.lang} a day. Your first lesson is ready.`;
  });

  back.addEventListener('click', () => cur > 0 && show(cur - 1));
</script>
