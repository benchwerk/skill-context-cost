document.getElementById('f').addEventListener('submit', e => {
  e.preventDefault();
  const d = new FormData(e.target);
  const day = new Date(d.get('date') + 'T00:00');
  if (day.getDay() === 1) { document.getElementById('ok').textContent = 'Sorry, we are closed on Mondays. Please pick another day.'; return; }
  const when = day.toLocaleDateString(undefined, { weekday: 'long', day: 'numeric', month: 'long' });
  document.getElementById('ok').textContent = `Thank you, ${d.get('name')}. We have your request for ${d.get('guests')} on ${when} at ${d.get('time')} and will confirm by email.`;
  e.target.reset();
});
