  Fri: [A("08:30","Aurelio Sandoval","Hygiene clean, Dr. Ruiz","Booked"),A("10:00","Henrietta Mbeki","Crown fitting, Dr. Vance","Booked"),A("13:30","Dmitri Falk","Extraction, Dr. Vance","Booked"),A("15:00","Cosima Rhodes","Check-up, Dr. Ruiz","Booked")],
  Sat: [A("09:00","Eliot Nakamura","Check-up, Dr. Ruiz","Booked"),A("10:30","Wilhelmina Grau","Hygiene clean, Dr. Ruiz","Booked")]
};
const TODAY = "Wed", SLOTS = 9;
let sel = TODAY;
const $ = id => document.getElementById(id);
$("date").textContent = "Wednesday, 7 October 2026. Two dentists on shift.";

function stats() {
  const d = data[TODAY], n = s => d.filter(x => x.s === s).length;
  const left = d.filter(x => x.s !== "Done").length;
  const rows = [[d.length, "Appointments today"], [n("Waiting") + n("In chair"), "In the building now"], [left, "Still to be seen"], [SLOTS - d.length, "Open slots left"]];
  $("stats").innerHTML = rows.map(([v, l], i) => `<div class="stat" style="--i:${i}"><b>${v}</b><span>${l}</span></div>`).join("");
}
function list() {
  const d = data[sel], first = d.findIndex(x => x.s === "In chair");
  $("title").textContent = sel === TODAY ? "Today's schedule" : sel + " schedule";
  $("list").innerHTML = d.length ? d.map((x, i) => `
    <div class="row${i === first && sel === TODAY ? " now" : ""}" style="--i:${i}">
      <span class="time">${x.t}</span><span class="who">${x.n}</span><span class="what">${x.w}</span>
      ${sel === TODAY ? `<button class="status" data-i="${i}" data-s="${x.s}" aria-label="${x.n}: ${x.s}. Advance status">${x.s}</button>`
        : `<span class="status" style="border:0;color:var(--muted);font-weight:400">${x.s === "Done" ? "Done" : "Booked"}</span>`}
    </div>`).join("") : `<p class="empty">No appointments booked. Use New booking to fill this day.</p>`;
}
function week() {
  const max = Math.max(...Object.values(data).map(a => a.length));
  $("week").innerHTML = Object.keys(data).map(k => `
    <button class="day" data-d="${k}" aria-pressed="${k === sel}">
      <span>${k}${k === TODAY ? " *" : ""}</span><i>${data[k].length}</i><em style="width:${data[k].length / max * 100}%"></em>
    </button>`).join("");
}
$("list").onclick = e => {
  const b = e.target.closest("[data-i]"); if (!b) return;
  const x = data[TODAY][b.dataset.i];
  x.s = FLOW[(FLOW.indexOf(x.s) + 1) % FLOW.length];
  stats(); list();
};
$("week").onclick = e => {
  const b = e.target.closest("[data-d]"); if (!b) return;
  sel = b.dataset.d; list(); week();
};
$("add").onclick = () => {
  const d = data[sel];
  if (d.length < SLOTS) { d.push(A("16:30", "Walk-in, name pending", "Check-up, Dr. Vance", "Booked")); stats(); list(); week(); }
};
stats(); list(); week();
