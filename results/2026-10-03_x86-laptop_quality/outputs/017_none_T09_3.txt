  const cmds = {
    brew: "brew install tidyup",
    cargo: "cargo install tidyup",
    curl: "curl -fsSL https://tidyup.dev/install.sh | sh"
  };
  let current = "brew";
  const out = document.getElementById("cmd");
  const tabs = document.querySelectorAll("[data-tab]");
  const show = k => {
    current = k;
    out.innerHTML = '<span class="p">$ </span>';
    out.append(cmds[k]);
    tabs.forEach(t => t.setAttribute("aria-selected", t.dataset.tab === k));
  };
  tabs.forEach(t => t.addEventListener("click", () => show(t.dataset.tab)));
  const btn = document.getElementById("copy");
  btn.addEventListener("click", async () => {
    try { await navigator.clipboard.writeText(cmds[current]); btn.textContent = "Copied!"; }
    catch { btn.textContent = "Press Ctrl+C"; }
    setTimeout(() => btn.textContent = "Copy", 1600);
  });
