<script>
  const date = document.getElementById('date');
  date.min = new Date().toISOString().split('T')[0];

  // Highlight today's row in the hours table
  const day = new Date().getDay();
  const map = { 1: 1, 2: 2, 3: 2, 4: 2, 5: 5, 6: 5, 0: 0 };
  document.querySelector(`tr[data-d="${map[day]}"]`).classList.add('today');

  document.getElementById('resForm').addEventListener('submit', e => {
    e.preventDefault();
    const f = new FormData(e.target);
    if (new Date(f.get('date') + 'T12:00').getDay() === 1) {
      return show('Sorry, we are closed on Mondays. Please choose another day.');
    }
    show(`Thank you, ${f.get('name')}! Your request for ${f.get('guests')} on ${f.get('date')} at ${f.get('time')} has been received.`);
    e.target.reset();
  });

  function show(text) {
    const m = document.getElementById('msg');
    m.textContent = text;
    m.hidden = false;
  }
</script>
