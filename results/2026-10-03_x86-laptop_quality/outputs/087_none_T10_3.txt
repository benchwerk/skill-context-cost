  });

  const tabs = document.querySelectorAll(".tabs button");

    tabs.forEach(t => t.setAttribute("aria-selected", t === tab));
    document.querySelectorAll(".day").forEach(d => d.classList.toggle("on", d.id === tab.dataset.day));
  }));

  document.querySelectorAll("[data-tier]").forEach(btn => btn.addEventListener("click", e => {
