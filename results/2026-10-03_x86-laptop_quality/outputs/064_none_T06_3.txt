<script>
  const date = document.getElementById('date');
  date.min = new Date().toISOString().split('T')[0];

  // Highlight today's row in the hours table
  const rows = document.querySelectorAll('#hoursTable tr');
  const map = [4, 0, 1, 1, 1, 2, 3]; // Sun..Sat -> row index
  rows[map[new Date().getDay()]].classList.add('today');

  document.getElementById('form').addEventListener('submit', e => {
    e.preventDefault();
    const f = new FormData(e.target);
    if (new Date(f.get('date') + 'T00:00').getDay() === 1) {
      document.getElementById('msg').textContent = 'Sorry, we are closed on Mondays. Please choose another day.';
      return;
    }
    document.getElementById('msg').textContent =
      `Thank you, ${f.get('name')}! Table for ${f.get('guests')} on ${f.get('date')} at ${f.get('time')} requested.`;
    e.target.reset();
  });
</script>
