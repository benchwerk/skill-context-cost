<script>
  const $ = (s, r = document) => r.querySelector(s), $$ = (s, r = document) => [...r.querySelectorAll(s)];
  let step = 1;
  const show = n => {
    step = n;
    $$('section').forEach(s => s.classList.toggle('active', +s.dataset.step === n));
    $$('.steps li').forEach((li, i) => li.classList.toggle('on', i < Math.min(n, 3)));
    $('#count').textContent = n > 3 ? 'All set' : `Step ${n} of 3`;
    const h = $(`section[data-step="${n}"] h1`); h.tabIndex = -1; h.focus({ preventScroll: true });
  };
  const validate = () => {
    const email = $('#email'), pw = $('#pw');
    $('#emailErr').textContent = email.validity.valid && email.value ? '' : 'Enter a valid email address.';
    $('#pwErr').textContent = pw.value.length >= 8 ? '' : 'Use at least 8 characters.';
    return !$('#emailErr').textContent && !$('#pwErr').textContent;
  };
  $$('[data-back]').forEach(b => b.onclick = () => show(step - 1));
  $$('input[type=radio]').forEach(r => r.onchange = () => $(`section[data-step="${step}"] .primary`).disabled = false);
  $('#account').onsubmit = e => e.preventDefault();
  $$('[data-next]').forEach(b => b.onclick = () => {
    if (step === 1 && !validate()) return;
    if (step === 3) {
      const lang = $('input[name=lang]:checked').value, goal = $('input[name=goal]:checked').value;
      $('#doneMin').textContent = goal + ' min';
      $('#doneTitle').textContent = `${lang}, ${goal} minutes a day.`;
      return show(4);
    }
    show(step + 1);
  });
  $('#account').addEventListener('keydown', e => { if (e.key === 'Enter') $('section.active [data-next]').click(); });
</script>
