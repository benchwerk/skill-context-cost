  const week = [["Mon", 8], ["Tue", 10], ["Wed", 7], ["Thu", 12], ["Fri", 9], ["Sat", 5]];
  const $ = id => document.getElementById(id);

  const today = new Date();
  $("date").textContent = today.toLocaleDateString("en-GB", { weekday: "long", day: "numeric", month: "long", year: "numeric" }) + " · Larkspur Dental";

  function render(filter) {
    const rows = appts.filter(a => filter === "all" || (filter === "done") === (a.s === "done"));
    $("list").innerHTML = rows.map((a, i) => `
      <li class="appt ${a.s === "done" ? "is-done" : ""}" style="animation-delay:${i * 40}ms">
        <span class="time">${a.t}</span>
        <div><div class="name">${a.n}</div><div class="what">${a.w}</div></div>
        <span class="tag ${a.s === "now" ? "now" : a.s}">${labels[a.s]}</span>
      </li>`).join("");
  }

  const done = appts.filter(a => a.s === "done").length;
  $("s-done").textContent = `${done} of ${appts.length}`;
  $("s-bar").style.width = (done / appts.length * 100) + "%";

  const max = Math.max(...week.map(d => d[1]));
  const todayIdx = (today.getDay() + 6) % 7;
  $("week").innerHTML = week.map(([d, n], i) => `
    <button class="day ${i === todayIdx ? "today" : ""}" style="--h:${n / max * 100}%" aria-label="${d}: ${n} appointments">
      <small>${n}</small><b style="animation-delay:${i * 60}ms"></b><span>${d}</span>
    </button>`).join("");

  document.querySelectorAll(".chip").forEach(c => c.addEventListener("click", () => {
    document.querySelectorAll(".chip").forEach(x => x.setAttribute("aria-pressed", x === c));
    render(c.dataset.f);
  }));
  render("all");
