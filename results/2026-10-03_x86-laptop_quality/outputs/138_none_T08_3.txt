<script>
  // Role filter
  const buttons = document.querySelectorAll('.filters button');
  const jobs = document.querySelectorAll('.jobs li');
  buttons.forEach(btn => btn.addEventListener('click', () => {
    buttons.forEach(b => b.setAttribute('aria-pressed', b === btn));
    jobs.forEach(li => { li.hidden = btn.dataset.team !== 'all' && li.dataset.team !== btn.dataset.team; });
  }));

  // Count-up for key numbers
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  const format = (el, v) => {
    const d = el.dataset, dec = String(d.count).includes('.') ? 2 : 0;
    el.textContent = (d.prefix || '') + (dec ? v.toFixed(dec) : Math.round(v).toLocaleString()) + (d.suffix || '');
  };
  const io = new IntersectionObserver(entries => entries.forEach(e => {
    if (!e.isIntersecting) return;
    io.unobserve(e.target);
    const el = e.target, end = parseFloat(el.dataset.count), t0 = performance.now();
    if (reduce) return format(el, end);
    const tick = now => {
      const p = Math.min((now - t0) / 1200, 1);
      format(el, end * (1 - Math.pow(1 - p, 3)));
      if (p < 1) requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  }), { threshold: .4 });
  document.querySelectorAll('[data-count]').forEach(el => io.observe(el));
</script>
