<script>
  const SLOTS = 12; // bookable slots per day
  const appts = [
    { t: "08:30", n: "Maya Hernandez", p: "Routine cleaning", fee: 140, done: true },
    { t: "09:15", n: "Tobias Reinholt", p: "Crown fitting", fee: 920, done: true },
    { t: "10:30", n: "Priya Nandakumar", p: "Fillings (2)", fee: 380, tag: "new" },
    { t: "11:15", n: "Samuel Adeyemi", p: "Root canal, visit 2", fee: 760 },
    { t: "13:00", n: "Elin Åkesson", p: "Whitening consult", fee: 90, tag: "new" },
    { t: "14:00", n: "Grace O'Donnell", p: "Check-up and X-ray", fee: 175 },
    { t: "15:30", n: "Louis Fabregat", p: "Orthodontic adjustment", fee: 210 },
  ];
  const week = [["Mon", 10], ["Tue", 9], ["Wed", 11], ["Thu", 8], ["Fri", 10], ["Sat", appts.length]];
  const $ = id => document.getElementById(id);
  const now = new Date();
  $("date").textContent = now.toLocaleDateString("en-GB", { weekday: "long", day: "numeric", month: "long" });
  const todayIdx = (now.getDay() + 6) % 7; // Mon = 0

  function render() {
    $("list").innerHTML = appts.map((a, i) => `
      <li class="appt ${a.done ? "done" : ""}">
        <span class="time">${a.t}</span>
        <div><div class="who">${a.n}${a.tag ? `<span class="tag ${a.tag}">${a.tag === "new" ? "New patient" : a.tag}</span>` : ""}</div>
        <div class="what">${a.p}</div></div>
        <button data-i="${i}" aria-pressed="${!!a.done}">${a.done ? "Seen" : "Mark seen"}</button>
      </li>`).join("");
    const done = appts.filter(a => a.done);
    $("s-done").textContent = done.length;
    $("s-left").textContent = appts.length - done.length;
    $("s-open").textContent = SLOTS - appts.length;
    $("s-rev").textContent = "$" + done.reduce((s, a) => s + a.fee, 0).toLocaleString();
  }

  $("list").addEventListener("click", e => {
    const b = e.target.closest("button");
    if (!b) return;
    appts[b.dataset.i].done = !appts[b.dataset.i].done;
    render();
    $("list").querySelector(`[data-i="${b.dataset.i}"]`).focus();
  });

  $("week").innerHTML = week.map(([d, c], i) => `
    <div class="day ${i === todayIdx ? "today" : ""}" role="img" aria-label="${d}: ${c} appointments">
      <div class="bar" style="height:${(c / SLOTS) * 100}%; animation-delay:${i * 60}ms"></div>
      <b>${c}</b><span>${d}</span>
    </div>`).join("");

  const total = week.reduce((s, [, c]) => s + c, 0);
  const quietest = week.reduce((m, d) => (d[1] < m[1] ? d : m));
  $("note").textContent = `${total} appointments booked. ${quietest[0]} is the quietest day with ${SLOTS - quietest[1]} open slots, a good place for recall calls.`;
  render();
</script>
