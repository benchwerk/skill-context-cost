  const cmds = {
    brew: "brew install fz",
    cargo: "cargo install fz-cli",
    curl: "curl -fsSL https://fz.example.dev/install.sh | sh"
  };
  const out = document.getElementById("cmd");
  const tabs = document.querySelectorAll("[role=tab]");
  tabs.forEach(t => t.addEventListener("click", () => {
    tabs.forEach(o => o.setAttribute("aria-selected", o === t));
    out.textContent = cmds[t.dataset.k];
  }));
  const btn = document.getElementById("copy");
  btn.addEventListener("click", async () => {
    try { await navigator.clipboard.writeText(out.textContent); btn.textContent = "Copied!"; }
    catch { btn.textContent = "Press Ctrl+C"; }
    setTimeout(() => btn.textContent = "Copy", 1500);
  });
