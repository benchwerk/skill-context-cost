<script>
const $=s=>document.querySelector(s), secs=[...document.querySelectorAll("section")], steps=[...document.querySelectorAll("ol li")];
let i=0;
function go(n){
  i=n; secs.forEach((s,k)=>s.hidden=k!==n);
  steps.forEach((li,k)=>{li.classList.toggle("done",k<n);k===n?li.setAttribute("aria-current","step"):li.removeAttribute("aria-current")});
  if(n===3)$("ol").hidden=true;
  secs[n].querySelector("h1").focus();
}
function fail(id,msg,input){$(id).textContent=msg;input&&input.setAttribute("aria-invalid","true");return false}
function clear(){document.querySelectorAll(".error").forEach(e=>e.textContent="");document.querySelectorAll("[aria-invalid]").forEach(e=>e.removeAttribute("aria-invalid"))}
const forms=document.querySelectorAll("form");
forms[0].addEventListener("submit",e=>{
  e.preventDefault(); clear(); let ok=true;
  const n=$("#name"),m=$("#email"),p=$("#pw");
  if(!n.value.trim())ok=fail("#e-name","Enter your first name.",n);
  if(!/^\S+@\S+\.\S+$/.test(m.value))ok=fail("#e-email","Enter a valid email address, like sam@example.com.",m);
  if(p.value.length<8)ok=fail("#e-pw","Password needs at least 8 characters.",p);
  if(!ok){document.querySelector("[aria-invalid]").focus();return}
  const b=e.submitter; b.disabled=true; b.textContent="Creating account...";
  setTimeout(()=>{b.disabled=false;b.textContent="Create account";go(1)},700);
});
forms[1].addEventListener("submit",e=>{
  e.preventDefault(); clear();
  if(!forms[1].lang.value)return fail("#e-lang","Pick a language to continue.");
  go(2);
});
forms[2].addEventListener("submit",e=>{
  e.preventDefault();
  $("#h3").textContent="Welcome, "+$("#name").value.trim()+".";
  $("#summary").textContent=forms[1].lang.value+", "+forms[2].goal.value+" minutes a day. Your first lesson is ready.";
  go(3);
});
document.querySelectorAll(".back").forEach(b=>b.addEventListener("click",()=>go(i-1)));
</script>
