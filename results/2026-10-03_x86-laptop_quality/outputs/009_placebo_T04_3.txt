  const appts = [
    ["8:30 AM", "Maria Santos", "Routine cleaning", "Okafor", "done"],
    ["9:15 AM", "Tomás Whitfield", "Crown fitting", "Lindqvist", "done"],
    ["10:00 AM", "Priya Raman", "Filling, lower left", "Okafor", "done"],
    ["10:45 AM", "Jonah Becker", "Root canal, stage 2", "Lindqvist", "done"],
    ["11:30 AM", "Ellen Park", "Whitening consult", "Okafor", "now"],
    ["12:15 PM", "Samir Haddad", "Extraction follow-up", "Lindqvist", "late"],
    ["2:30 PM", "Chloe Ferreira", "Child exam", "Okafor", "next"],
    ["3:15 PM", "Daniel Osei", "Night guard fitting", "Lindqvist", "next"],
    ["5:00 PM", "Ruth Calloway", "Denture adjustment", "Okafor", "next"]
  ];
  const labels = { done: "Completed", now: "In chair", late: "Running late", next: "Upcoming" };
  const week = [8, 11, 9, 12, 10, 5]; // Mon–Sat
  const names = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat"];
  const notes = ["Light day, good for recall calls.", "Busy, two hygiene blocks.", "Steady.", "Busiest day, double-check lab cases.", "Steady, early close at 4 PM.", "Half day, 9 AM to 1 PM."];

  const now = new Date();
  document.getElementById("today").textContent =
    now.toLocaleDateString(undefined, { weekday: "long", month: "long", day: "numeric" }) + " · " + appts.length + " patients";

  const done = appts.filter(a => a[4] === "done").length;
  sTotal.textContent = appts.length;
  sDone.textContent = done;
  sPct.textContent = Math.round(done / appts.length * 100) + "%";

  function renderList(filter) {
    list.innerHTML = appts
      .filter(a => filter === "all" || a[3] === filter)
      .map(([t, name, what, dr, st]) =>
        `<li class="appt ${st === "done" ? "done" : ""}"><time>${t}</time>
         <div><b>${name}</b><span class="sub">${what} · Dr. ${dr}</span></div>
         <span class="pill ${st === "next" ? "" : st}">${labels[st]}</span></li>`)
      .join("");
  }
  document.querySelectorAll(".tabs button").forEach(b => b.addEventListener("click", () => {
    document.querySelectorAll(".tabs button").forEach(x => x.setAttribute("aria-pressed", x === b));
    renderList(b.dataset.f);
  }));
  renderList("all");

  const todayIdx = (now.getDay() + 6) % 7; // Monday = 0
  const max = Math.max(...week);
  week.forEach((n, i) => {
    const d = document.createElement("button");
    d.className = "day" + (i === todayIdx ? " today" : "");
    d.setAttribute("aria-label", `${names[i]}: ${n} appointments`);
    d.innerHTML = `<em>${n}</em><i style="height:${n / max * 100}px"></i>${names[i]}`;
    d.addEventListener("click", () => weekNote.textContent = `${names[i]}: ${n} booked. ${notes[i]}`);
    document.getElementById("week").appendChild(d);
  });
  weekNote.textContent = `Total this week: ${week.reduce((a, b) => a + b)} appointments. Tap a bar for detail.`;
