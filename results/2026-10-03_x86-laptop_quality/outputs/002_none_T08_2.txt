<script>
  const jobs = [
    { title: "Senior Backend Engineer", team: "Engineering", loc: "Remote (EU/US)" },
    { title: "Staff Security Engineer", team: "Engineering", loc: "London" },
    { title: "Product Designer", team: "Design", loc: "Remote" },
    { title: "Credit Risk Analyst", team: "Risk", loc: "New York" },
    { title: "Compliance Manager", team: "Risk", loc: "London" },
    { title: "Customer Success Lead", team: "Operations", loc: "Lagos" }
  ];
  const teams = ["All", ...new Set(jobs.map(j => j.team))];
  const filters = document.querySelector(".filters");
  const list = document.querySelector(".jobs");

  function render(team) {
    const shown = jobs.filter(j => team === "All" || j.team === team);
    list.innerHTML = shown.map(j =>
      `<li><div><strong>${j.title}</strong><small>${j.team} · ${j.loc}</small></div>` +
      `<a class="apply" href="mailto:careers@ledgerly.example?subject=${encodeURIComponent(j.title)}">Apply</a></li>`
    ).join("") || '<li class="empty">No open roles in this team right now.</li>';
    filters.querySelectorAll("button").forEach(b => b.setAttribute("aria-pressed", b.textContent === team));
  }

  teams.forEach(t => {
    const b = document.createElement("button");
    b.type = "button";
    b.textContent = t;
    b.addEventListener("click", () => render(t));
    filters.appendChild(b);
  });
  render("All");

  // Count-up animation for key numbers, skipped for reduced-motion users
  if (!matchMedia("(prefers-reduced-motion: reduce)").matches) {
    const io = new IntersectionObserver(entries => entries.forEach(e => {
      if (!e.isIntersecting) return;
      io.unobserve(e.target);
      const el = e.target, end = +el.dataset.count, dec = +(el.dataset.dec || 0);
      const pre = el.dataset.prefix || "", suf = el.dataset.suffix || "", t0 = performance.now();
      (function tick(now) {
        const p = Math.min((now - t0) / 1200, 1), v = end * (1 - Math.pow(1 - p, 3));
        el.textContent = pre + v.toLocaleString("en-US", { minimumFractionDigits: dec, maximumFractionDigits: dec }) + suf;
        if (p < 1) requestAnimationFrame(tick);
      })(t0);
    }), { threshold: .5 });
    document.querySelectorAll("[data-count]").forEach(el => io.observe(el));
  }
</script>
