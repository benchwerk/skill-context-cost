  const buttons = { monthly: document.getElementById('monthly'), yearly: document.getElementById('yearly') };
  function setPeriod(p) {
    for (const k in buttons) buttons[k].setAttribute('aria-pressed', k === p);
    const key = p === 'yearly' ? 'y' : 'm';
    document.querySelectorAll('span[data-m]').forEach(el => el.textContent = el.dataset[key]);
    document.querySelectorAll('div.note[data-m]').forEach(el => el.textContent = el.dataset[key]);
  }
  buttons.monthly.onclick = () => setPeriod('monthly');
  buttons.yearly.onclick = () => setPeriod('yearly');
