<script>
  const CAPACITY = 12;
  const appts = [
    { t: "08:30", name: "Maya Okafor",      tx: "Cleaning & exam",     dr: "Dr. Lindqvist", s: "done" },
    { t: "09:15", name: "Tomás Herrera",    tx: "Composite filling",   dr: "Dr. Lindqvist", s: "done" },
    { t: "10:00", name: "Priya Raman",      tx: "Crown preparation",   dr: "Dr. Abara",     s: "now"  },
    { t: "11:00", name: "Jonas Eklund",     tx: "Check-up",            dr: "Dr. Lindqvist", s: "next" },
    { t: "13:00", name: "Aisha Rahman",     tx: "Whitening consult",   dr: "Dr. Abara",     s: "next" },
    { t: "14:00", name: "Louis Bernard",    tx: "Root canal, part 2",  dr: "Dr. Abara",     s: "next" },
    { t: "15:30", name: "Hannah Kowalski",  tx: "Invisalign review",   dr: "Dr. Lindqvist", s: "next" },
    { t: "16:15", name: "Sam Whitfield",    tx: "Emergency: chipped tooth", dr: "Dr. Lindqvist", s: "next" }
  ];
  const week = [["Mon", 11], ["Tue", 12], ["Wed", 9], ["Thu", 8], ["Fri", 5], ["Sat", 3]];
  const todayIdx = (new Date().getDay() + 6) % 7;
  const label = { done: "Completed", now: "In chair", next: "Upcoming" };
  const $ = id => document.getElementById(id);

  function render() {
    $("list").innerHTML = appts.map((a, i) => `
      <div class="row" tabindex="0" data-s="${a.s}" data-i="${i}">
        <span class="time">${a.t}</span>
        <div class="who"><strong>${a.name}</strong><span>${a.tx} · ${a.dr}</span></div>
        <span class="pill">${label[a.s]}</span>
      </div>`).join("");

    const done = appts.filter(a => a.s === "done").length;
    const left = appts.filter(a => a.s === "next").length;
    const weekTotal = week.reduce((n, d) => n + d[1], 0);
    const free = week.reduce((n, d) => n + CAPACITY - d[1], 0);
    $("count").textContent = `${done} of ${appts.length} seen`;
    $("stats").innerHTML = [
      ["Today", appts.length, "appointments"],
      ["Waiting", left, left ? "still to arrive" : "all seen"],
      ["Week booked", weekTotal, `of ${CAPACITY * week.length} slots`],
      ["Open slots", free, "across the week"]
    ].map(([k, v, s]) => `<div class="card stat"><small>${k}</small><b>${v}</b><span>${s}</span></div>`).join("");
  }

  $("week").innerHTML = week.map(([d, n], i) => `
    <div class="day${i === todayIdx ? " today" : ""}${n >= CAPACITY ? " full" : ""}">
      <b>${d}</b>
      <div class="track"><div class="fill" style="width:${(n / CAPACITY) * 100}%"></div></div>
      <i>${n}/${CAPACITY}</i>
    </div>`).join("");
  const busiest = week.reduce((a, b) => (b[1] > a[1] ? b : a));
  const quietest = week.reduce((a, b) => (b[1] < a[1] ? b : a));
  $("note").textContent = `${busiest[0]} is fully booked. ${quietest[0]} has the most room for walk-ins.`;
  $("date").textContent = new Date().toLocaleDateString(undefined, { weekday: "long", day: "numeric", month: "long", year: "numeric" });

  function advance() {
    const now = appts.findIndex(a => a.s === "now");
    if (now > -1) appts[now].s = "done";
    const next = appts.find(a => a.s === "next");
    if (next) next.s = "now";
    $("advance").disabled = !appts.some(a => a.s !== "done");
    render();
  }
  $("advance").addEventListener("click", advance);
  $("list").addEventListener("keydown", e => { if (e.key === "Enter" && e.target.dataset.i) e.target.click(); });
  render();
</script>
