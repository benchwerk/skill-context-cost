<script>
  const steps = [...document.querySelectorAll('#track li')];
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  let timer;
  function run() {
    clearInterval(timer);
    let i = -1;
    const tick = () => {
      i++;
      steps.forEach((s, k) => { s.classList.toggle('done', k <= i); s.classList.toggle('now', k === i); });
      if (i >= 2) clearInterval(timer);
    };
    steps.forEach(s => s.classList.remove('done', 'now'));
    if (reduce) { i = 1; tick(); return; }
    timer = setInterval(tick, 1100); tick();
  }
  document.querySelectorAll('.chip').forEach(c => c.addEventListener('click', () => {
    document.querySelectorAll('.chip').forEach(o => o.setAttribute('aria-pressed', o === c));
    bean.textContent = c.dataset.n; roast.textContent = c.dataset.r + ' roast - 340g'; eta.textContent = c.dataset.e;
    run();
  }));
  run();
  document.getElementById('join').addEventListener('submit', e => {
    e.preventDefault();
    msg.textContent = 'You are on the list. We will email ' + email.value + ' on launch day.';
    e.target.reset();
  });
  const io = new IntersectionObserver(es => es.forEach(x => { if (x.isIntersecting) { x.target.classList.add('in'); io.unobserve(x.target); } }), { threshold: .2 });
  document.querySelectorAll('.reveal').forEach((el, i) => { el.style.transitionDelay = (i % 4) * 80 + 'ms'; io.observe(el); });
</script>
