<script>
  const $ = (s, r = document) => r.querySelector(s), $$ = (s, r = document) => [...r.querySelectorAll(s)];
  const steps = $$('.step'), bars = $$('.bars i'), back = $('.back'), form = $('form');
  let cur = 0;

  function go(n) {
    cur = n;
    steps.forEach((s, i) => s.classList.toggle('active', i === n));
    bars.forEach((b, i) => b.classList.toggle('on', i <= n));
    $('.count').textContent = `${n + 1} / 3`;
    back.hidden = n === 0;
    document.title = `Parla – ${steps[n].dataset.title}`;
    $('h1', steps[n]).tabIndex = -1;
    $('h1', steps[n]).focus({ preventScroll: true });
  }

  function fail(id, msg) {
    const el = $('#' + id);
    el.setAttribute('aria-invalid', !!msg);
    $(`[data-for=${id}]`).textContent = msg;
    return !msg;
  }

  function validateAccount() {
    const okEmail = fail('email', /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test($('#email').value) ? '' : 'Enter a valid email address.');
    const okPw = fail('pw', $('#pw').value.length >= 8 ? '' : 'Use at least 8 characters.');
    return okEmail && okPw;
  }

  $('.toggle').onclick = e => {
    const pw = $('#pw'), show = pw.type === 'password';
    pw.type = show ? 'text' : 'password';
    e.target.textContent = show ? 'Hide' : 'Show';
    e.target.setAttribute('aria-label', show ? 'Hide password' : 'Show password');
  };

  $$('[data-next]').forEach(b => b.onclick = () => {
    if (cur === 0 && !validateAccount()) return $('[aria-invalid=true]').focus();
    go(cur + 1);
  });
  $$('input[name=lang]').forEach(r => r.onchange = () => $('[data-next]', steps[1]).disabled = false);
  back.onclick = () => go(cur - 1);
  $$('#email, #pw').forEach(i => i.oninput = () => i.getAttribute('aria-invalid') === 'true' && validateAccount());

  form.onsubmit = e => {
    e.preventDefault();
    const lang = $('input[name=lang]:checked').value, goal = $('input[name=goal]:checked').value;
    $('header').hidden = true;
    form.innerHTML = `<div class="done" tabindex="-1"><div class="big">🎉</div>
      <h1>You're all set</h1>
      <p class="lede">${lang}, ${goal} minutes a day. Your first lesson is ready.</p></div>`;
    $('.done').focus();
  };
</script>
