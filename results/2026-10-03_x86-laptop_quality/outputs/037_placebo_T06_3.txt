  const today = new Date();
  document.querySelector(`.hours tr[data-d="${today.getDay()}"]`).classList.add("today");

  f.date.min = today.toISOString().split("T")[0];
  f.addEventListener("submit", e => {
    e.preventDefault();
    const d = new Date(f.date.value + "T12:00");
    if (d.getDay() === 1) { msg.textContent = "We're closed on Mondays — please pick another evening."; return; }
    const day = d.toLocaleDateString("en", { weekday: "long", month: "long", day: "numeric" });
    msg.textContent = `Thank you, ${f.name.value.split(" ")[0]}. We'll hold a table for ${f.guests.value} on ${day} at ${f.time.value}.`;
    f.reset(); f.guests.value = 2;
  });
