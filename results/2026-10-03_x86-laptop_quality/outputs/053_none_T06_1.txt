<script>
  const date = document.getElementById('date');
  const today = new Date().toISOString().split('T')[0];
  date.min = today;

  const dow = new Date().getDay();
  document.querySelectorAll('.hours tr').forEach(r => {
    const d = +r.dataset.d;
    const match = d === dow || (d === 2 && dow >= 2 && dow <= 4) || (d === 5 && dow >= 5 && dow <= 6);
    if (match) r.classList.add('today');
  });

  document.getElementById('f').addEventListener('submit', e => {
    e.preventDefault();
    const f = e.target;
    if (new Date(date.value + 'T12:00').getDay() === 1) {
      document.getElementById('msg').textContent = 'Sorry, we are closed on Mondays. Please pick another day.';
      return;
    }
    document.getElementById('msg').textContent =
      `Thank you, ${f.name.value}! Table for ${f.guests.value} on ${date.value} at ${f.time.value} requested.`;
    f.reset();
  });
</script>
