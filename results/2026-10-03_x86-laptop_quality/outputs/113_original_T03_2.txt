const f=document.getElementById('f'),ok=document.getElementById('ok'),msgs={n:'Please enter your names.',e:'Please enter a valid email address.',d:'Please choose a date.',m:'Please tell me a little about the day.'};
f.addEventListener('submit',ev=>{
  ev.preventDefault();
  let bad=false;
  for(const el of f.elements){
    if(!el.name)continue;
    const miss=!el.checkValidity();
    el.setAttribute('aria-invalid',miss);
    el.parentElement.querySelector('.err').textContent=miss?msgs[el.name]:'';
    bad=bad||miss;
  }
  if(bad){f.querySelector('[aria-invalid=true]').focus();return}
  const b=f.querySelector('button');
  b.disabled=true;b.textContent='Sending...';
  setTimeout(()=>{f.hidden=true;ok.hidden=false;ok.focus()},800);
});
const io=new IntersectionObserver(es=>es.forEach(e=>{if(e.isIntersecting){e.target.classList.add('in');io.unobserve(e.target)}}),{threshold:.15});
